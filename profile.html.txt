        if (errorBox) {
          errorBox.classList.remove("show");
        }

      function toast(message, type = "success") {
        if (!toastContainer) return;

        const item = document.createElement("div");
        item.className = `toast ${type}`;
        item.textContent = message;

        toastContainer.appendChild(item);

        setTimeout(() => {
          item.style.opacity = "0";
          item.style.transform = "translateY(10px)";

          setTimeout(() => item.remove(), 250);
        }, 3200);

      function safeString(value) {
        if (value === null || value === undefined) return "";
        return String(value).trim();
      }

      function firstValue(...values) {
        for (const value of values) {
          if (value !== null && value !== undefined && safeString(value) !== "") {
            return value;
          }
        }

        return "";
      }

      function getInitials(name) {
        const value = safeString(name);

        if (!value) return "؟";

        const parts = value.split(/\s+/).filter(Boolean);

        if (parts.length === 1) {
          return parts[0].slice(0, 2);
        }

        return (parts[0][0] || "") + (parts[1][0] || "");
      }

      function normalizeStudent(student) {
        if (!student) return null;

        const source = { ...student };

        const name = firstValue(
          source.name,
          source.fullName,
          source.studentName,
          source.displayName,
          source.username
        );

        const phone = firstValue(
          source.phone,
          source.mobile,
          source.phoneNumber
        );

        const email = firstValue(
          source.email,
          source.emailAddress
        );

        const grade = firstValue(
          source.grade,
          source.year,
          source.level,
          source.class,
          source.studyYear
        );

        const studentId = firstValue(
          source.studentId,
          source.id,
          source.uid,
          source.userId,
          source.key,
          source.studentKey
        );

        const status = firstValue(
          source.status,
          source.accountStatus
        );

        return {
          ...source,
          name,
          phone,
          email,
          grade,
          studentId,
          status
        };
      }

      function isApproved(student) {
        const status = safeString(
          firstValue(student?.status, student?.accountStatus)
        ).toLowerCase();

        if (!status) return true;

        return [
          "approved",
          "active",
          "مقبول",
          "نشط",
          "فعال"
        ].includes(status);
      }

      function fillProfile(student) {
        currentStudent = normalizeStudent(student);

        if (!currentStudent) {
          throw new Error("لم يتم العثور على بيانات الطالب.");
        }

        const name = currentStudent.name || "طالب";
        const phone = currentStudent.phone || "غير مسجل";
        const email = currentStudent.email || "";
        const grade = currentStudent.grade || "غير محدد";
        const studentId = currentStudent.studentId || currentStudentKey || "—";

        profileName.textContent = name;
        profilePhone.textContent = phone;
        topUserName.textContent = name;

        quickGrade.textContent = grade;
        quickEmail.textContent = email || "غير مسجل";
        quickStudentId.textContent = studentId;

        nameInput.value = name;
        phoneInput.value = phone;
        gradeInput.value = grade;
        emailInput.value = email;
        studentIdInput.value = studentId;

        const initials = getInitials(name);

        profileAvatar.textContent = initials;
        topAvatar.textContent = initials;

        if (isApproved(currentStudent)) {
          profileStatus.innerHTML = `
            <span class="status-dot"></span>
            حساب نشط
          `;
        } else {
          profileStatus.innerHTML = `
            <span class="status-dot"></span>
            ${safeString(currentStudent.status) || "حالة الحساب"}
          `;
        }

        originalValues = {
          name: name,
          grade: grade === "غير محدد" ? "" : grade,
          email: email
        };
      }

      async function getCurrentStudent() {
        let student = null;

        try {
          if (
            window.FirebaseFeatures &&
            typeof window.FirebaseFeatures.getProfile === "function"
          ) {
            student = await window.FirebaseFeatures.getProfile();
          }
        } catch (error) {
          console.warn("FirebaseFeatures.getProfile failed:", error);
        }

        if (!student) {
          try {
            if (
              window.FirebaseAuth &&
              typeof window.FirebaseAuth.getCurrentUser === "function"
            ) {
              student = await window.FirebaseAuth.getCurrentUser();
            }
          } catch (error) {
            console.warn("FirebaseAuth.getCurrentUser failed:", error);
          }
        }

        if (!student) {
          try {
            const raw = localStorage.getItem("currentUser");

            if (raw) {
              student = JSON.parse(raw);
            }
          } catch (error) {
            console.warn("Could not parse currentUser:", error);
          }
        }

        if (!student) {
          try {
            const raw = localStorage.getItem("student");

            if (raw) {
              student = JSON.parse(raw);
            }
          } catch (error) {
            console.warn("Could not parse student:", error);
          }
        }

        if (!student) {
          try {
            if (
              window.FirebaseDatabase &&
              typeof window.FirebaseDatabase.students?.getCurrent === "function"
            ) {
              student = await window.FirebaseDatabase.students.getCurrent();
            }
          } catch (error) {
            console.warn("FirebaseDatabase.students.getCurrent failed:", error);
          }
        }

        if (!student) {
          return null;
        }

        try {
          if (
            window.FirebaseDatabase &&
            typeof window.FirebaseDatabase.students?.getCurrent === "function"
          ) {
            const databaseStudent =
              await window.FirebaseDatabase.students.getCurrent();

            if (databaseStudent) {
              student = databaseStudent;
            }
          }
        } catch (error) {
          console.warn("Database student refresh failed:", error);
        }

        return student;
      }

      async function determineStudentKey(student) {
        const direct = firstValue(
          student?.studentKey,
          student?.key,
          student?.id,
          student?.studentId,
          student?.uid,
          student?.userId,
          student?.phone
        );

        if (direct) {
          try {
            if (
              window.FirebaseDatabase &&
              typeof window.FirebaseDatabase.students?.getByKey === "function"
            ) {
              const found =
                await window.FirebaseDatabase.students.getByKey(String(direct));

              if (found) {
                return String(direct);
              }
            }
          } catch (_) {}
        }

        try {
          if (
            window.FirebaseDatabase &&
            typeof window.FirebaseDatabase.students?.find === "function"
          ) {
            const found = await window.FirebaseDatabase.students.find(student);

            if (found) {
              return firstValue(
                found.studentKey,
                found.key,
                found.id,
                found.studentId,
                found.uid,
                found.userId,
                found.phone
              );
            }
          }
        } catch (_) {}

        return direct || null;
      }

      async function waitForFirebase() {
        const timeout = 12000;
        const start = Date.now();

        while (Date.now() - start < timeout) {
          const ready =
            window.FirebaseCoreReady &&

          showError(message);
          toast(message, "error");

          saveButton.disabled = false;
          saveButton.textContent = "حفظ التغييرات";
        }
