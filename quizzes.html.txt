            function escapeHtml(value) {
                return String(value == null ? "" : value)
                    .replace(/&/g, "&amp;")
                    .replace(/</g, "&lt;")
                    .replace(/>/g, "&gt;")
                    .replace(/"/g, "&quot;")
                    .replace(/'/g, "&#039;");
            }

            function safeString(value, fallback) {
                if (
                    value === null ||
                    value === undefined
                ) {
                    return fallback || "";
                }

                const result = String(value).trim();

                return result || (fallback || "");
            }

            function numberValue(value, fallback) {
                const n = Number(value);

                return Number.isFinite(n)
                    ? n
                    : (fallback || 0);
            }

            function normalizeTerm(value) {
                if (
                    value === null ||
                    value === undefined ||
                    value === ""
                ) {
                    return "";
                }

                const raw = String(value)
                    .trim()
                    .toLowerCase();

                const map = {
                    "term1": "1",
                    "term01": "1",
                    "first": "1",
                    "first term": "1",
                    "الترم الاول": "1",
                    "الترم الأول": "1",

                    "term2": "2",
                    "term02": "2",
                    "second": "2",
                    "second term": "2",
                    "الترم الثاني": "2",

                    "term3": "3",
                    "term03": "3",
                    "third": "3",
                    "third term": "3",
                    "الترم الثالث": "3",

                    "term4": "4",
                    "term04": "4",
                    "fourth": "4",
                    "fourth term": "4",
                    "الترم الرابع": "4"
                };

                if (map[raw]) {
                    return map[raw];
                }

                const match = raw.match(/[1-4]/);

                return match
                    ? match[0]
                    : raw;
            }

            function isPublished(quiz) {
                if (!quiz) {
                    return false;
                }

                if (
                    quiz.published === false ||
                    quiz.active === false
                ) {
                    return false;
                }

                const status =
                    safeString(quiz.status)
                        .toLowerCase();

                if (
                    status === "draft" ||
                    status === "hidden" ||
                    status === "disabled"
                ) {
                    return false;
                }

                return true;
            }

            function isPaid(quiz) {
                if (!quiz) {
                    return false;
                }

                const values = [
                    quiz.paid,
                    quiz.isPaid,
                    quiz.premium,
                    quiz.requiresPayment,
                    quiz.paymentRequired
                ];

                for (const value of values) {

                    if (
                        value === true ||
                        String(value).toLowerCase() === "true"
                    ) {
                        return true;
                    }

                    if (
                        String(value).toLowerCase() === "paid" ||
                        String(value).toLowerCase() === "premium"
                    ) {
                        return true;
                    }
                }

                const price = Number(
                    quiz.price ??
                    quiz.amount ??
                    quiz.cost
                );

                return Number.isFinite(price) && price > 0;
            }

            function getQuizId(quiz) {
                return safeString(
                    quiz.id ||

                if (Array.isArray(quiz.questions)) {
                    return quiz.questions.length;

                if (
                    quiz.questions &&

                    return Object.keys(

                    quiz.durationMinutes ??

            function normalizeQuiz(quiz, fallbackId) {
                const source = quiz || {};

                const normalized = {
                    ...source,

                    id: getQuizId(source) || fallbackId || "",

                    title:
                        getQuizTitle(source),

                    description:
                        getQuizDescription(source),

                    subject:
                        getSubjectName(source),

                    subjectId:
                        safeString(
                            source.subjectId ||
                            source.subjectID ||

                        ),

                    term:

                        ),

                    grade:
                        safeString(

                            source.year ||
                            source.level
                        ),

                    duration:
                        getDuration(source),

                    passScore:
                        getPassScore(source),

                    questionCount:
                        getQuestionCount(source),

                    paid:
                        isPaid(source),

                    published:
                        isPublished(source),

                    order:
                        getOrder(source),

                    createdAt:
                        getCreatedAt(source)
                };

                return normalized;
            }

            function normalizeQuizCollection(data) {
                const list = [];

                if (Array.isArray(data)) {

                    data.forEach(function (item, index) {

                        if (!item) {
                            return;
                        }

                        list.push(
                            normalizeQuiz(
                                item,
                                String(index)
                            )
                        );

                    });

                    return list;
                }

                if (
                    data &&
                    typeof data === "object"
                ) {

                    Object.keys(data).forEach(function (key) {

                        const item = data[key];

                        if (!item) {
                            return;
                        }

                        list.push(
                            normalizeQuiz(
                                item,
                                key
                            )
                        );

                    });
                }

                return list;
            }

            function uniqueQuizzes(list) {
                const map = new Map();

                list.forEach(function (quiz) {

                    const id =
                        getQuizId(quiz) ||
                        (
                            getQuizTitle(quiz) +
                            "|" +
                            getSubjectName(quiz)
                        );

                    if (!map.has(id)) {
                        map.set(id, quiz);
                    }

                });

                return Array.from(map.values());
            }

            function sortQuizzes(list) {
                return list.slice().sort(function (a, b) {

                    const orderA =
                        numberValue(a.order, 999999);

                    const orderB =
                        numberValue(b.order, 999999);

                    if (orderA !== orderB) {
                        return orderA - orderB;
                    }

                    return (
                        numberValue(b.createdAt, 0) -
                        numberValue(a.createdAt, 0)
                    );
                });
            }

            /* =========================================

            function waitForModules(
                timeout
            ) {
                timeout =
                    timeout || 12000;

                return new Promise(
                    function (resolve, reject) {

                        const started =
                            Date.now();

                        function check() {

                            const ready =
                                window.FirebaseCoreReady &&
                                window.FirebaseAuthReady &&
                                window.FirebaseDatabaseReady &&
                                window.FirebaseContentReady &&
                                window.FirebaseFeaturesReady;

                            if (ready) {
                                resolve();
                                return;
                            }

                            if (
                                Date.now() -
                                started >
                                timeout
                            ) {
                                reject(
                                    new Error(
                                        "Firebase modules timeout"
                                    )
                                );

                                return;
                            }

                            setTimeout(
                                check,
                                100
                            );
                        }

                        check();
                    }
                );
            }

            async function loadQuizzes() {

                state.loading = true;

                try {

                    await waitForModules();

                    let quizzes = [];

                    const Content =
                        window.FirebaseContent;

                    const DB =
                        window.FirebaseDatabase;

                    /*
                     * الأولوية:
                     * FirebaseContent
                     * ثم FirebaseFeatures
                     * ثم Database
                     */

                    if (
                        Content &&
                        typeof Content.getQuizzes ===
                        "function"
                    ) {

                        quizzes =
                            await Content.getQuizzes();

                    } else if (
                        Features &&
                        typeof Features.getQuizzes ===
                        "function"
                    ) {

                        quizzes =
                            await Features.getQuizzes();

                    } else if (
                        DB &&
                        DB.quizzes &&
                        typeof DB.quizzes.getAll ===
                        "function"
                    ) {

                        quizzes =
                            await DB.quizzes.getAll();

                    } else {

                        throw new Error(

                        );
                    }

                    quizzes =
                        normalizeQuizCollection(
                            quizzes
                        );

                    quizzes =
                        quizzes.filter(
                            isPublished
                        );

                    quizzes =
                        uniqueQuizzes(
                            quizzes
                        );

                    quizzes =
                        sortQuizzes(
                            quizzes
                        );

                    state.quizzes =
                        quizzes;

                    state.loading = false;

                    applyFilters();

                    hideLoader();

                } catch (error) {

                    console.error(

                    );

                    state.loading = false;
                    state.error = error;

                    hideLoader();

                    elements.skeletonGrid.hidden =
                        true;

                    elements.quizGrid.hidden =
                        true;

                    elements.emptyState.hidden =
                        true;

                    elements.errorState.hidden =
                        false;

                    elements.errorText.textContent =
                        getFriendlyError(
                            error
                        );

                    elements.sectionCount.textContent =
                        "تعذر التحميل";
                }
            }

            function getFriendlyError(error) {

                if (!error) {
                    return "حدث خطأ غير معروف.";
                }

                const code =
                    safeString(
                        error.code
                    ).toLowerCase();

                if (
                    code.includes("permission")
                ) {
                    return "لا توجد صلاحية للوصول إلى الاختبارات حاليًا.";
                }

                if (
                    code.includes("network")
                ) {
                    return "تأكد من اتصال الإنترنت وحاول مرة أخرى.";
                }

                if (
                    error.message
                ) {
                    return error.message;
                }

                return "تعذر تحميل الاختبارات. حاول مرة أخرى.";
            }

            /* =========================================
               FILTERS
            ========================================== */

            function applyFilters() {

                const search =
                    state.search
                        .trim()
                        .toLowerCase();

                let result =
                    state.quizzes.slice();

                if (
                    state.type === "free"
                ) {

                    result =
                        result.filter(
                            function (quiz) {
                                return !quiz.paid;
                            }
                        );

                } else if (
                    state.type === "paid"
                ) {

                    result =
                        result.filter(
                            function (quiz) {
                                return quiz.paid;
                            }
                        );
                }

                if (
                    state.term !== "all"
                ) {

                    result =
                        result.filter(
                            function (quiz) {

                                return (
                                    normalizeTerm(
                                        quiz.term
                                    ) ===
                                    state.term
                                );

                            }
                        );
                }

                if (search) {

                    result =
                        result.filter(
                            function (quiz) {

                                const searchable =
                                    [
                                        quiz.title,
                                        quiz.description,
                                        quiz.subject,
                                        quiz.subjectId,
                                        quiz.grade,
                                        quiz.term
                                    ]
                                        .join(" ")
                                        .toLowerCase();

                                return searchable
                                    .includes(
                                        search
                                    );
                            }
                        );
                }

                state.filtered =
                    result;

                renderQuizzes();
            }

            function renderQuizCard(

                const title =
                    escapeHtml(

                    );

                const description =
                    escapeHtml(

                    );

                const subject =

                const questionCount =

                const duration =

            function bindQuizButtons() {

                elements.quizGrid
                    .querySelectorAll(
                        "[data-start-quiz]"
                    )
                    .forEach(
                        function (button) {

                            button.addEventListener(
                                "click",
                                function () {

                                    const id =
                                        button.getAttribute(
                                            "data-start-quiz"
                                        );

                                    startQuiz(
                                        id
                                    );
                                }
                            );

                        }
                    );
            }

            async function startQuiz(
                quizId
            ) {

                if (!quizId) {
                    showToast(
                        "تعذر تحديد الاختبار.",
                        "error"
                    );

                    return;
                }

                try {

                    const Auth =
                        window.FirebaseAuth;

                    let identity =
                        null;

                    if (
                        Auth &&
                        typeof Auth.getCurrentUser ===
                        "function"
                    ) {
                        identity =
                            Auth.getCurrentUser();
                    }

                    if (!identity) {

                        try {

                            const DB =
                                window.FirebaseDatabase;

                            if (
                                DB &&
                                typeof DB.getCurrentUser ===
                                "function"
                            ) {
                                identity =
                                    DB.getCurrentUser();
                            }

                        } catch (ignore) {}
                    }

                    if (!identity) {

                        showToast(
                            "سجّل الدخول أولًا لبدء الاختبار.",
                            "error"
                        );

                        setTimeout(
                            function () {
                                window.location.href =
                                    "login.html?next=" +
                                    encodeURIComponent(
                                        "quizzes.html"
                                    );
                            },
                            700
                        );

                        return;
                    }

                    /*
                     * لا ننشئ محاولة في قاعدة البيانات
                     * هنا إلا إذا كانت طبقة Features
                     * مهيأة لبدء الاختبار.
                     *
                     * وإلا ننتقل لصفحة الاختبار
                     * مع الـ quizId فقط.
                     */

                    const Features =
                        window.FirebaseFeatures;

                    if (
                        Features &&
                        typeof Features.startQuiz ===
                        "function"
                    ) {

                        try {

                            const attempt =
                                await Features.startQuiz(
                                    quizId
                                );

                            const attemptId =
                                attempt &&
                                (
                                    attempt.id ||
                                    attempt.attemptId ||
                                    attempt.key
                                );

                            let url =
                                "quiz.html?id=" +
                                encodeURIComponent(
                                    quizId
                                );

                            if (
                                attemptId
                            ) {
                                url +=
                                    "&attemptId=" +
                                    encodeURIComponent(
                                        attemptId
                                    );
                            }

                            window.location.href =
                                url;

                            return;

                        } catch (startError) {

                            /*
                             * لو startQuiz غير متوافق
                             * مع نسخة صفحة الاختبار،
                             * نسمح للصفحة نفسها بإدارة البداية.
                             */

                            console.warn(
                                "startQuiz fallback:",
                                startError
                            );
                        }
                    }

                    window.location.href =
                        "quiz.html?id=" +
                        encodeURIComponent(
                            quizId
                        );

                } catch (error) {

                    console.error(
                        "Start quiz error:",
                        error
                    );

                    showToast(
                        "تعذر فتح الاختبار. حاول مرة أخرى.",
                        "error"
                    );
                }
            }

            /* =========================================
               TYPE FILTER EVENTS
            ========================================== */

            if (
                elements.typeFilters
            ) {

                elements.typeFilters
                    .querySelectorAll(
                        "[data-type]"
                    )
                    .forEach(
                        function (button) {

                            button.addEventListener(
                                "click",
                                function () {

                                    elements.typeFilters
                                        .querySelectorAll(
                                            "[data-type]"
                                        )
                                        .forEach(
                                            function (item) {
                                                item.classList.remove(
                                                    "active"
                                                );
                                            }
                                        );

                                    button.classList.add(
                                        "active"
                                    );

                                    state.type =
                                        button.getAttribute(
                                            "data-type"
                                        ) ||
                                        "all";

                                    applyFilters();
                                }
                            );

                        }
                    );
            }

            /* =========================================
               TERM EVENTS
            ========================================== */

            if (
                elements.terms
            ) {

                elements.terms
                    .querySelectorAll(
                        "[data-term]"
                    )
                    .forEach(
                        function (button) {

                            button.addEventListener(
                                "click",
                                function () {

                                    elements.terms
                                        .querySelectorAll(
                                            "[data-term]"
                                        )
                                        .forEach(
                                            function (item) {
                                                item.classList.remove(
                                                    "active"
                                                );
                                            }
                                        );

                                    button.classList.add(
                                        "active"
                                    );

                                    state.term =
                                        button.getAttribute(
                                            "data-term"
                                        ) ||
                                        "all";

                                    applyFilters();
                                }
                            );

                        }
                    );
            }

            /* =========================================
               SEARCH
            ========================================== */

            if (
                elements.searchInput
            ) {

                let searchTimer =
                    null;

                elements.searchInput.addEventListener(
                    "input",
                    function () {

                        clearTimeout(
                            searchTimer
                        );

                        searchTimer =
                            setTimeout(
                                function () {

                                    state.search =
                                        elements.searchInput.value ||
                                        "";

                                    applyFilters();

                                },
                                120
                            );
                    }
                );
            }

            /* =========================================
               RETRY
            ========================================== */

            if (
                elements.retryButton
            ) {

                elements.retryButton.addEventListener(
                    "click",
                    loadQuizzes
                );
            }

            /* =========================================
               AUTH STATE
            ========================================== */

            function setupAuthState() {

                const Auth =
                    window.FirebaseAuth;

                if (
                    Auth &&
                    typeof Auth.onAuthStateChanged ===
                    "function"
                ) {

                    try {

                        Auth.onAuthStateChanged(
                            function (user) {

                                if (!user) {
                                    return;
                                }

                            }
                        );

                    } catch (error) {
                        console.warn(
                            "Auth listener unavailable:",
                            error
                        );
                    }
                }
            }

            /* =========================================
               YEAR
            ========================================== */

            if (
                elements.currentYear
            ) {
                elements.currentYear.textContent =
                    new Date().getFullYear();
            }

            /* =========================================
               GLOBAL API
            ========================================== */

            window.CapQuizzes = {

                state: state,

                load: loadQuizzes,

                reload: loadQuizzes,

                applyFilters:
                    applyFilters,

                getQuizzes:
                    function () {
                        return state.quizzes.slice();
                    },

                getFilteredQuizzes:
                    function () {
                        return state.filtered.slice();
                    },

                getQuizById:
                    function (id) {

                        return state.quizzes.find(
                            function (quiz) {
                                return (
                                    getQuizId(quiz) ===
                                    String(id)
                                );
                            }
                        ) || null;
                    },

                startQuiz:
                    startQuiz

            };

            /* =========================================
               INITIALIZE
            ========================================== */

            document.addEventListener(
                "DOMContentLoaded",
                function () {

                    setupAuthState();

                    loadQuizzes();

                }
            );

        })();
    </script>
