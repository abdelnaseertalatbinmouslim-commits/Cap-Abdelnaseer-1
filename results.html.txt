               BASIC HELPERS
               ===================================================== */

            safeString(value, fallback = "") {

                if (
                    value === null ||
                    value === undefined
                ) {
                    return fallback;
                }

                return String(value).trim() || fallback;
            },


            escapeHtml(value) {

                return this.safeString(value)
                    .replace(/&/g, "&amp;")
                    .replace(/</g, "&lt;")
                    .replace(/>/g, "&gt;")
                    .replace(/"/g, "&quot;")
                    .replace(/'/g, "&#039;");
            },


            normalizeDigits(value) {

                return this.safeString(value)
                    .replace(/[٠-٩]/g, digit =>
                        String(
                            "٠١٢٣٤٥٦٧٨٩".indexOf(digit)
                        )
                    )
                    .replace(/[۰-۹]/g, digit =>
                        String(
                            "۰۱۲۳۴۵۶۷۸۹".indexOf(digit)
                        )
                    );
            },


            firstValue(...values) {

                for (const value of values) {

                    if (
                        value !== undefined &&
                        value !== null &&
                        String(value).trim() !== ""
                    ) {
                        return value;
                    }

                }

                return null;
            },


            toArray(value) {

                if (Array.isArray(value)) {
                    return value;
                }

                if (
                    value &&
                    typeof value === "object"
                ) {

                    return Object.keys(value)
                        .map(key => {

                            const item = value[key];

                            if (
                                item &&
                                typeof item === "object"
                            ) {
                                return {
                                    ...item,
                                    _key: item._key || key
                                };
                            }

                            return {
                                value: item,
                                _key: key
                            };

                        });

                }

                return [];
            },


            getTimestamp(value) {

                if (
                    value === null ||
                    value === undefined ||
                    value === ""
                ) {
                    return null;
                }

                if (
                    typeof value === "object" &&
                    value.seconds
                ) {
                    return Number(value.seconds) * 1000;
                }

                if (
                    typeof value === "object" &&
                    value.timestamp
                ) {
                    return this.getTimestamp(
                        value.timestamp
                    );
                }

                if (typeof value === "number") {

                    if (value < 10000000000) {
                        return value * 1000;
                    }

                    return value;
                }

                const number = Number(value);

                if (!Number.isNaN(number)) {

                    if (number < 10000000000) {
                        return number * 1000;
                    }

                    return number;
                }

                const parsed = Date.parse(
                    String(value)
                );

                return Number.isNaN(parsed)
                    ? null
                    : parsed;
            },


            formatDate(value) {

                const timestamp =
                    this.getTimestamp(value);

                if (!timestamp) {
                    return "غير مسجل";
                }

                try {

                    return new Intl.DateTimeFormat(
                        "ar-EG",
                        {
                            year: "numeric",
                            month: "short",
                            day: "numeric",
                            hour: "numeric",
                            minute: "2-digit"
                        }
                    ).format(
                        new Date(timestamp)
                    );

                } catch (error) {

                    return new Date(timestamp)
                        .toLocaleString("ar-EG");

                }

            },

            /* =====================================================
               NORMALIZE RESULT
               ===================================================== */

            normalizeResult(item, source = "result") {

                if (
                    !item ||
                    typeof item !== "object"
                ) {
                    return null;
                }

                const id =
                    this.firstValue(
                        item.id,
                        item.resultId,
                        item.attemptId,
                        item._key,
                        item.key
                    );


                const quizId =
                    this.firstValue(
                        item.quizId,
                        item.quizID,
                        item.quiz,
                        item.examId,
                        item.testId
                    );


                const title =
                    this.firstValue(
                        item.title,
                        item.quizTitle,
                        item.quizName,
                        item.name,
                        item.examTitle,
                        quizId
                            ? `اختبار ${quizId}`
                            : "اختبار"
                    );


                const subject =
                    this.firstValue(
                        item.subject,
                        item.subjectName,
                        item.subjectTitle,
                        item.subjectId
                    );


                const term =
                    this.firstValue(
                        item.term,
                        item.semester,
                        item.termId,
                        item.yearTerm
                    );


                const totalQuestionsRaw =
                    this.firstValue(
                        item.totalQuestions,
                        item.questionsCount,
                        item.total,
                        item.maxQuestions
                    );


                const answeredQuestionsRaw =
                    this.firstValue(
                        item.answeredQuestions,
                        item.answered,
                        item.answersCount
                    );


                const scoreRaw =
                    this.firstValue(
                        item.score,
                        item.mark,
                        item.points,
                        item.gradeScore,
                        item.obtainedScore
                    );


                const percentageRaw =
                    this.firstValue(
                        item.percentage,
                        item.percent,
                        item.scorePercentage,
                        item.progress
                    );


                let totalQuestions =
                    Number(totalQuestionsRaw);

                if (
                    !Number.isFinite(totalQuestions) ||
                    totalQuestions < 0
                ) {
                    totalQuestions = null;
                }


                let answeredQuestions =
                    Number(answeredQuestionsRaw);

                if (
                    !Number.isFinite(answeredQuestions) ||
                    answeredQuestions < 0
                ) {
                    answeredQuestions = null;
                }


                let score =
                    Number(scoreRaw);

                if (
                    !Number.isFinite(score)
                ) {
                    score = null;
                }


                let percentage =
                    Number(percentageRaw);

                if (
                    !Number.isFinite(percentage)
                ) {
                    percentage = null;
                }


                /*
                 * IMPORTANT:
                 *
                 * We calculate a percentage ONLY when
                 * both score and total questions are actually
                 * present in Firebase.
                 *
                 * We never invent a score.
                 */

                if (
                    percentage === null &&
                    score !== null &&
                    totalQuestions !== null &&
                    totalQuestions > 0
                ) {

                    percentage =
                        (score / totalQuestions) * 100;

                }


                if (percentage !== null) {

                    percentage =
                        Math.max(
                            0,
                            Math.min(
                                100,
                                percentage
                            )
                        );

                }


                const statusRaw =
                    this.firstValue(
                        item.status,
                        item.state
                    );


                const normalizedStatus =
                    this.safeString(
                        statusRaw
                    ).toLowerCase();


                const submittedAt =
                    this.firstValue(
                        item.submittedAt,
                        item.completedAt,
                        item.finishedAt,
                        item.updatedAt
                    );


                const startedAt =
                    this.firstValue(
                        item.startedAt,
                        item.createdAt
                    );


                const completed =
                    Boolean(
                        item.completed === true ||
                        item.isCompleted === true ||
                        item.submitted === true ||
                        submittedAt ||
                        [
                            "completed",
                            "complete",
                            "submitted",
                            "finished",
                            "done",
                            "مكتمل",
                            "مكتملة"
                        ].includes(normalizedStatus)
                    );


                let passed = null;


                if (
                    item.passed === true ||
                    item.isPassed === true
                ) {

                    passed = true;

                } else if (
                    item.passed === false ||
                    item.isPassed === false
                ) {

                    passed = false;

                }


                const passScoreRaw =
                    this.firstValue(
                        item.passScore,
                        item.passingScore
                    );


                const passScore =
                    Number(passScoreRaw);


                if (
                    passed === null &&
                    percentage !== null &&
                    Number.isFinite(passScore)
                ) {

                    passed =
                        percentage >=
                        Number(passScore);

                }


                const studentKey =
                    this.firstValue(
                        item.studentKey,
                        item.studentId,
                        item.uid,
                        item.userId,
                        item.phone,
                        item.studentPhone
                    );


                return {

                    ...item,

                    _normalized: true,

                    _source: source,

                    _id: this.safeString(
                        id,
                        `${source}_${Math.random()
                            .toString(36)
                            .slice(2)}`
                    ),

                    _quizId: this.safeString(
                        quizId
                    ),

                    _title: this.safeString(
                        title,
                        "اختبار"
                    ),

                    _subject: this.safeString(
                        subject,
                        "غير محدد"
                    ),

                    _term: this.safeString(
                        term
                    ),

                    _totalQuestions:
                        totalQuestions,

                    _answeredQuestions:
                        answeredQuestions,

                    _score:
                        score,

                    _percentage:
                        percentage,

                    _passed:
                        passed,

                    _completed:
                        completed,

                    _studentKey:
                        this.safeString(
                            studentKey
                        ),

                    _submittedAt:
                        submittedAt,

                    _startedAt:
                        startedAt,

                    _date:
                        this.getTimestamp(
                            submittedAt
                        ) ||
                        this.getTimestamp(
                            startedAt
                        ) ||
                        0

                };

            },


            /* =====================================================
               DEDUPLICATION
               ===================================================== */

            getResultUniqueKey(result) {

                if (!result) {
                    return "";
                }

                const attemptId =
                    this.safeString(
                        result.attemptId
                    );

                if (attemptId) {
                    return `attempt:${attemptId}`;
                }

                if (result._id) {

                    return [
                        result._source,
                        result._id
                    ].join(":");

                }

                return [
                    result._quizId,
                    result._title,
                    result._date,
                    result._studentKey
                ].join("|");

            },


            deduplicateResults(results) {

                const map = new Map();

                for (const result of results) {

                    if (!result) {
                        continue;
                    }

                    const key =
                        this.getResultUniqueKey(
                            result
                        );

                    if (!key) {
                        continue;
                    }

                    const existing =
                        map.get(key);

                    if (!existing) {

                        map.set(
                            key,
                            result
                        );

                        continue;
                    }


                    /*
                     * Prefer the richer record.
                     */

                    const existingScore =
                        existing._score !== null ||
                        existing._percentage !== null;

                    const newScore =
                        result._score !== null ||
                        result._percentage !== null;


                    if (
                        newScore &&
                        !existingScore
                    ) {

                        map.set(
                            key,
                            result
                        );

                        continue;

                    }


                    if (
                        result._date >
                        existing._date
                    ) {

                        map.set(
                            key,
                            {
                                ...existing,
                                ...result
                            }
                        );

                    }

                }

                return Array.from(
                    map.values()
                ).sort(
                    (a, b) =>
                        (b._date || 0) -
                        (a._date || 0)
                );

            },

            async loadResults() {

                const DB =
                    window.FirebaseDatabase;

                const Features =
                    window.FirebaseFeatures;


                const student =

                    this.state.studentKey;

                    "function"
                ) {

                            await Features
                                .getResults();

                            this.toArray(
                                featureResults
                            );

                        array.forEach(item => {

                            const normalized =
                                this.normalizeResult(
                                    item,
                                    "features"
                                );

                            if (normalized) {
                                collected.push(
                                    normalized
                                );
                            }

                        });

                    } catch (error) {

                        console.warn(
                            "Features results failed:",
                            error
                        );

                    }

                }


                /*
                 * /results
                 */

                if (
                    DB &&
                    DB.results &&
                    typeof DB.results.getCombined ===
                    "function"
                ) {

                    try {

                        let data = null;

                        try {

                            data =
                                await DB.results
                                    .getCombined(
                                        student
                                    );

                        } catch (firstError) {

                            if (studentKey) {

                                data =
                                    await DB.results
                                        .getCombined(
                                            studentKey
                                        );

                            } else {

                                throw firstError;

                            }

                        }


                        const array =
                            this.toArray(data);

                        array.forEach(item => {

                            const normalized =
                                this.normalizeResult(
                                    item,
                                    "results"
                                );

                            if (normalized) {
                                collected.push(
                                    normalized
                                );
                            }

                        });

                    } catch (error) {

                        console.warn(
                            "Main results failed:",
                            error
                        );

                    }

                }


                /*
                 * Legacy /results direct API
                 */

                if (
                    DB &&
                    DB.results &&
                    typeof DB.results.getAll ===
                    "function"
                ) {

                    try {

                        let data = null;

                        try {

                            data =
                                await DB.results
                                    .getAll(
                                        student
                                    );

                        } catch (firstError) {

                            if (studentKey) {

                                data =
                                    await DB.results
                                        .getAll(
                                            studentKey
                                        );

                            } else {

                                throw firstError;

                            }

                        }


                        const array =
                            this.toArray(data);

                        array.forEach(item => {

                            const normalized =
                                this.normalizeResult(
                                    item,
                                    "results-all"
                                );

                            if (normalized) {
                                collected.push(
                                    normalized
                                );
                            }

                        });

                    } catch (error) {

                        console.warn(
                            "Results getAll failed:",
                            error
                        );

                    }

                }


                /*
                 * Legacy /quiz_results
                 */

                if (
                    DB &&
                    DB.results &&
                    typeof DB.results.getLegacy ===
                    "function"
                ) {

                    try {

                        let data = null;

                        try {

                            data =
                                await DB.results
                                    .getLegacy(
                                        student
                                    );

                        } catch (firstError) {

                            if (studentKey) {

                                data =
                                    await DB.results
                                        .getLegacy(
                                            studentKey
                                        );

                            } else {

                                throw firstError;

                            }

                        }


                        const array =
                            this.toArray(data);

                        array.forEach(item => {

                            const normalized =
                                this.normalizeResult(
                                    item,
                                    "quiz_results"
                                );

                            if (normalized) {
                                collected.push(
                                    normalized
                                );
                            }

                        });

                    } catch (error) {

                        console.warn(
                            "Legacy results failed:",
                            error
                        );

                    }

                }


                /*
                 * Quiz attempts.
                 *
                 * This is especially important because
                 * the old project stored attempts under:
                 *
                 * /quizAttempts/{studentKey}
                 */

                if (
                    DB &&
                    DB.attempts &&
                    typeof DB.attempts.getAll ===
                    "function"
                ) {

                    try {

                        let data = null;

                        if (studentKey) {

                            try {

                                data =
                                    await DB.attempts
                                        .getAll(
                                            studentKey
                                        );

                            } catch (firstError) {

                                try {

                                    data =
                                        await DB.attempts
                                            .getAll(
                                                student
                                            );

                                } catch (secondError) {

                                    console.warn(
                                        "Attempts student lookup failed:",
                                        secondError
                                    );

                                }

                            }

                        }


                        if (data) {

                            const array =
                                this.toArray(data);

                            array.forEach(item => {

                                const normalized =
                                    this.normalizeResult(
                                        item,
                                        "quizAttempts"
                                    );

                                if (normalized) {
                                    collected.push(
                                        normalized
                                    );
                                }

                            });

                        }

                    } catch (error) {

                        console.warn(
                            "Quiz attempts failed:",
                            error
                        );

                    }

                }


                /*
                 * Final fallback:
                 * old DB layer may expose getQuizAttempts.
                 */

                if (
                    DB &&
                    typeof DB.getQuizAttempts ===
                    "function"
                ) {

                    try {

                        let data = null;

                        if (studentKey) {

                            data =
                                await DB
                                    .getQuizAttempts(
                                        studentKey
                                    );

                        }


                        const array =
                            this.toArray(data);

                        array.forEach(item => {

                            const normalized =
                                this.normalizeResult(
                                    item,
                                    "quizAttempts-legacy"
                                );

                            if (normalized) {
                                collected.push(
                                    normalized
                                );
                            }

                        });

                    } catch (error) {

                        console.warn(
                            "Legacy getQuizAttempts failed:",
                            error
                        );

                    }

                }


                /*
                 * Remove records that clearly belong
                 * to another student when a student key
                 * exists in the record.
                 */

                const filtered =
                    collected.filter(
                        result => {

                            if (
                                !result._studentKey ||
                                !studentKey
                            ) {
                                return true;
                            }

                            return (
                                String(
                                    result._studentKey
                                ) ===
                                String(studentKey)
                            );

                        }
                    );


                return this.deduplicateResults(
                    filtered
                );

            },


            /* =====================================================
               LOAD PAGE
               ===================================================== */

            async load() {

                if (this.state.loading) {
                    return;
                }

                this.state.loading = true;

                this.showLoader();


                try {

                    this.state.student =
                        await this.getCurrentStudent();


                    if (!this.state.student) {

                        window.location.replace(
                            "login.html?redirect=" +
                            encodeURIComponent(
                                "results.html"
                            )
                        );

                        return;

                    }


                    this.state.studentKey =
                        this.getStudentKey(
                            this.state.student
                        );


                    this.updateStudentUI();


                    const results =
                        await this.loadResults();


                    this.state.allResults =
                        results;


                    this.updateStats();


                    this.applyFilters();


                    this.hideLoader();


                } catch (error) {

                    console.error(
                        "Results page error:",
                        error
                    );


                    this.showError(
                        this.getReadableError(
                            error
                        )
                    );


                    this.hideLoader();

                } finally {

                    this.state.loading = false;

                }

            },


            /* =====================================================
               ERROR
               ===================================================== */

            getReadableError(error) {

                if (!error) {
                    return "تعذر تحميل النتائج.";
                }

                const code =
                    this.safeString(
                        error.code
                    ).toLowerCase();


                if (
                    code.includes("permission")
                ) {

                    return "لا توجد صلاحية كافية لقراءة النتائج حاليًا.";

                }


                if (
                    code.includes("network")
                ) {

                    return "تحقق من اتصال الإنترنت وحاول مرة أخرى.";

                }


                if (
                    error.message
                ) {

                    return this.safeString(
                        error.message,
                        "تعذر تحميل النتائج."
                    );

                }


                return "تعذر تحميل النتائج حاليًا.";

            },


            showError(message) {

                const errorText =
                    document.getElementById(
                        "errorText"
                    );

                if (errorText) {
                    errorText.textContent =
                        message;
                }

                this.showOnly(
                    "errorState"
                );

            },


            /* =====================================================
               STUDENT UI
               ===================================================== */

            updateStudentUI() {

                const student =
                    this.state.student;

                if (!student) {
                    return;
                }


                const name =
                    this.safeString(
                        this.firstValue(
                            student.name,
                            student.studentName,
                            student.fullName,
                            "الطالب"
                        ),
                        "الطالب"
                    );


                const studentName =
                    document.getElementById(
                        "studentName"
                    );

                if (studentName) {
                    studentName.textContent =
                        name;
                }


                const avatar =
                    document.getElementById(
                        "studentAvatar"
                    );

                if (avatar) {

                    const firstChar =
                        name
                            .replace(/\s+/g, " ")
                            .trim()
                            .charAt(0);

                    avatar.textContent =
                        firstChar || "ط";

                }

            },


            /* =====================================================
               STATS
               ===================================================== */

            updateStats() {

                const results =
                    this.state.allResults;


                const total =
                    results.length;


                const completed =
                    results.filter(
                        result =>
                            result._completed
                    ).length;


                const graded =
                    results.filter(
                        result =>
                            result._score !== null ||
                            result._percentage !== null
                    ).length;


                const totalElement =
                    document.getElementById(
                        "totalResults"
                    );

                const completedElement =
                    document.getElementById(
                        "completedResults"
                    );

                const gradedElement =
                    document.getElementById(
                        "gradedResults"
                    );


                if (totalElement) {
                    totalElement.textContent =
                        total;
                }


                if (completedElement) {
                    completedElement.textContent =
                        completed;
                }


                if (gradedElement) {
                    gradedElement.textContent =
                        graded;
                }

            },


            /* =====================================================
               FILTERING
               ===================================================== */

            setFilter(filter) {

                this.state.filter =
                    this.safeString(
                        filter,
                        "all"
                    );


                document
                    .querySelectorAll(
                        ".filter-button"
                    )
                    .forEach(button => {

                        button.classList.toggle(
                            "active",
                            button.dataset.filter ===
                            this.state.filter
                        );

                    });


                this.applyFilters();

            },


            setSearch(value) {

                this.state.search =
                    this.safeString(
                        value
                    ).toLowerCase();

                this.applyFilters();

            },


            applyFilters() {

                let results =
                    [...this.state.allResults];


                if (
                    this.state.filter ===
                    "completed"
                ) {

                    results =
                        results.filter(
                            result =>
                                result._completed
                        );

                }


                if (
                    this.state.filter ===
                    "graded"
                ) {

                    results =
                        results.filter(
                            result =>
                                result._score !== null ||
                                result._percentage !== null
                        );

                }


                if (
                    this.state.filter ===
                    "pending"
                ) {

                    results =
                        results.filter(
                            result =>
                                !result._completed
                        );

                }


                const search =
                    this.state.search;


                if (search) {

                    results =
                        results.filter(
                            result => {

                                const haystack = [
                                    result._title,
                                    result._subject,
                                    result._term,
                                    result._quizId
                                ]
                                    .join(" ")
                                    .toLowerCase();

                                return haystack
                                    .includes(search);

                            }
                        );

                }


                this.state.filteredResults =
                    results;


                this.render();

            },


            /* =====================================================
               RESULT STATUS
               ===================================================== */

            getStatus(result) {

                if (!result._completed) {

                    return {
                        text: "غير مكتملة",
                        className: "warning"
                    };

                }


                if (
                    result._passed === true
                ) {

                    return {
                        text: "ناجح",
                        className: "success"
                    };

                }


                if (
                    result._passed === false
                ) {

                    return {
                        text: "غير ناجح",
                        className: "danger"
                    };

                }


                if (
                    result._score !== null ||
                    result._percentage !== null
                ) {

                    return {
                        text: "مكتملة",
                        className: "success"
                    };

                }


                return {
                    text: "مكتملة",
                    className: "neutral"
                };

            },


            getScoreText(result) {

                if (
                    result._score !== null &&
                    result._totalQuestions !== null
                ) {

                    return `${this.formatNumber(
                        result._score
                    )} / ${this.formatNumber(
                        result._totalQuestions
                    )}`;

                }


                if (
                    result._score !== null
                ) {

                    return this.formatNumber(
                        result._score
                    );

                }


                if (
                    result._percentage !== null
                ) {

                    return `${this.formatNumber(
                        result._percentage
                    )}%`;

                }


                return "غير مسجل";

            },


            getPercentageText(result) {

                if (
                    result._percentage === null
                ) {

                    return "غير متاح";

                }


                return `${this.formatNumber(
                    result._percentage
                )}%`;

            },


            formatNumber(value) {

                if (
                    value === null ||
                    value === undefined
                ) {
                    return "—";
                }


                const number =
                    Number(value);


                if (!Number.isFinite(number)) {
                    return this.safeString(
                        value,
                        "—"
                    );
                }


                if (
                    Number.isInteger(number)
                ) {
                    return String(number);
                }


                return number
                    .toFixed(2)
                    .replace(/\.00$/, "");

            },


            /* =====================================================
               RENDER
               ===================================================== */

            render() {

                const results =
                    this.state.filteredResults;


                if (
                    !this.state.allResults.length
                ) {

                    this.showOnly(
                        "emptyState"
                    );

                    return;

                }


                if (!results.length) {

                    const grid =
                        document.getElementById(
                            "resultsGrid"
                        );

                    if (grid) {

                        grid.innerHTML = `
                            <div class="empty-state"
                                 style="grid-column:1/-1;">
                                <div class="state-icon">🔎</div>
                                <h2 class="state-title">
                                    مفيش نتائج مطابقة
                                </h2>
                                <p class="state-text">
                                    جرّب تغيّر الفلتر أو كلمة البحث.
                                </p>
                            </div>
                        `;

                        this.showOnly(
                            "resultsGrid"
                        );

                    }

                    return;

                }


                const grid =
                    document.getElementById(
                        "resultsGrid"
                    );


                if (!grid) {
                    return;
                }


                grid.innerHTML =
                    results
                        .map(result =>
                            this.renderResultCard(
                                result
                            )
                        )
                        .join("");


                this.showOnly(
                    "resultsGrid"
                );

            },


            renderResultCard(result) {

                const status =
                    this.getStatus(
                        result
                    );


                const title =
                    this.escapeHtml(
                        result._title
                    );


                const subject =
                    this.escapeHtml(
                        result._subject
                    );


                const term =
                    result._term
                        ? this.escapeHtml(
                            `ترم ${result._term}`
                        )
                        : "غير محدد";


                const total =
                    result._totalQuestions !== null
                        ? this.formatNumber(
                            result._totalQuestions
                        )
                        : "—";


                const answered =
                    result._answeredQuestions !== null
                        ? this.formatNumber(
                            result._answeredQuestions
                        )
                        : "—";


                const score =
                    this.escapeHtml(
                        this.getScoreText(
                            result
                        )
                    );


                const percentage =
                    this.escapeHtml(
                        this.getPercentageText(
                            result
                        )
                    );


                const date =
                    this.escapeHtml(
                        this.formatDate(
                            result._submittedAt ||
                            result._startedAt
                        )
                    );


                const quizId =
                    this.safeString(
                        result._quizId
                    );


                let actionHtml = "";


                if (quizId) {

                    const params =
                        new URLSearchParams();

                    params.set(
                        "id",
                        quizId
                    );


                    if (
                        result.attemptId ||
                        result._id
                    ) {

                        params.set(
                            "attemptId",
                            this.safeString(
                                result.attemptId ||
                                result._id
                            )
                        );

                    }


                    actionHtml = `
                        <a
                            class="result-action"
                            href="quiz.html?${params.toString()}"
                        >
                            مراجعة الاختبار
                            <span>←</span>
                        </a>
                    `;

                } else {

                    actionHtml = `
                        <span
                            class="result-action"
                            style="opacity:.55;cursor:default;"
                        >
                            لا يوجد رابط للاختبار
                        </span>
                    `;

                }


                return `
                    <article class="result-card">

                        <div class="result-head">

                            <div class="result-title-wrap">

                                <h2 class="result-title">
                                    ${title}
                                </h2>

                                <div class="result-subject">
                                    ${subject}
                                </div>

                            </div>

                            <span
                                class="status-badge ${status.className}"
                            >
                                ${status.text}
                            </span>

                        </div>


                        <div class="result-divider"></div>


                        <div class="result-info-grid">

                            <div class="result-info">

                                <div class="info-label">
                                    الترم
                                </div>

                                <div class="info-value">
                                    ${term}
                                </div>

                            </div>


                            <div class="result-info">

                                <div class="info-label">
                                    الأسئلة
                                </div>

                                <div class="info-value">
                                    ${total}
                                </div>

                            </div>


                            <div class="result-info">

                                <div class="info-label">
                                    المجاب
                                </div>

                                <div class="info-value">
                                    ${answered}
                                </div>

                            </div>


                            <div class="result-info">

                                <div class="info-label">
                                    الدرجة
                                </div>

                                <div class="info-value">
                                    ${score}
                                </div>

                            </div>


                            <div class="result-info">

                                <div class="info-label">
                                    النسبة
                                </div>

                                <div class="info-value">
                                    ${percentage}
                                </div>

                            </div>


                            <div class="result-info">

                                <div class="info-label">
                                    التاريخ
                                </div>

                                <div
                                    class="info-value"
                                    title="${date}"
                                >
                                    ${date}
                                </div>

                            </div>

                        </div>


                        <div class="score-area">

                            <div class="score-main">

                                <div class="score-circle">
                                    ${
                                        result._percentage !== null
                                            ? this.formatNumber(
                                                result._percentage
                                            ) + "%"
                                            : "—"
                                    }
                                </div>

                                <div>

                                    <div class="score-text-title">
                                        الدرجة المسجلة
                                    </div>

                                    <div class="score-text-value">
                                        ${score}
                                    </div>

                                </div>

                            </div>


                            ${actionHtml}

                        </div>

                    </article>
                `;

            },


            /* =====================================================
               EVENTS
               ===================================================== */

            bindEvents() {

                document
                    .querySelectorAll(
                        ".filter-button"
                    )
                    .forEach(button => {

                        button.addEventListener(
                            "click",
                            () => {

                                this.setFilter(
                                    button.dataset.filter
                                );

                            }
                        );

                    });


                const searchInput =
                    document.getElementById(
                        "searchInput"
                    );


                if (searchInput) {

                    searchInput.addEventListener(
                        "input",
                        event => {

                            this.setSearch(
                                event.target.value
                            );

                        }
                    );

                }


                const retryButton =
                    document.getElementById(
                        "retryButton"
                    );


                if (retryButton) {

                    retryButton.addEventListener(
                        "click",
                        async () => {

                            this.showOnly(
                                "skeletonGrid"
                            );

                            await this.load();

                        }
                    );

                }


                const mobileButton =
                    document.getElementById(
                        "mobileMenuButton"
                    );


                const mobileNav =
                    document.getElementById(
                        "mobileNav"
                    );


                if (
                    mobileButton &&
                    mobileNav
                ) {

                    mobileButton.addEventListener(
                        "click",
                        () => {

                            mobileNav.classList.toggle(
                                "open"
                            );

                        }
                    );


                    mobileNav
                        .querySelectorAll("a")
                        .forEach(link => {

                            link.addEventListener(
                                "click",
                                () => {

                                    mobileNav
                                        .classList
                                        .remove(
                                            "open"
                                        );

                                }
                            );

                        });

                }


                document.addEventListener(
                    "click",
                    event => {

                        if (
                            !mobileNav ||
                            !mobileButton
                        ) {
                            return;
                        }


                        if (
                            !mobileNav.contains(
                                event.target
                            ) &&
                            !mobileButton.contains(
                                event.target
                            )
                        ) {

                            mobileNav.classList.remove(
                                "open"
                            );

                        }

                    }
                );

            },


            /* =====================================================
               FIREBASE READINESS
               ===================================================== */

            async waitForFirebase(
                timeout = 15000
            ) {

                const start =
                    Date.now();


                while (
                    Date.now() - start <
                    timeout
                ) {

                    if (
                        window.FirebaseCoreReady &&
                        window.FirebaseAuthReady &&
                        window.FirebaseDatabaseReady &&
                        window.FirebaseContentReady &&
                        window.FirebaseFeaturesReady
                    ) {

                        return true;

                    }


                    await new Promise(
                        resolve =>
                            setTimeout(
                                resolve,
                                80
                            )
                    );

                }


                /*
                 * Some pages may still work if the features
                 * wrapper is not flagged but the DB is ready.
                 */

                return Boolean(
                    window.FirebaseDatabase &&
                    window.FirebaseAuth
                );

            },


            /* =====================================================
               INIT
               ===================================================== */

            async init() {

                const year =
                    document.getElementById(
                        "currentYear"
                    );

                if (year) {

                    year.textContent =
                        new Date()
                            .getFullYear();

                }


                this.bindEvents();


                try {

                    await this.waitForFirebase();


                    await this.load();

                } catch (error) {

                    console.error(
                        "Results initialization failed:",
                        error
                    );


                    this.showError(
                        this.getReadableError(
                            error
                        )
                    );

                    this.hideLoader();

                }

            }

        };


        /* =========================================================
           PUBLIC API
           ========================================================= */

        window.CapResults =
            ResultsPage;


        /* =========================================================
           START
           ========================================================= */

        document.addEventListener(
            "DOMContentLoaded",
            () => {

                ResultsPage.init();

            }
        );

    </script>
