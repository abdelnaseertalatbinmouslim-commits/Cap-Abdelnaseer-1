            function normalizeArabicNumbers(value) {

                return String(value || "")

                    .replace(
                        /[٠-٩]/g,
                        function (digit) {

                            return String(
                                "٠١٢٣٤٥٦٧٨٩"
                                .indexOf(digit)
                            );

                        }
                    )

                    .replace(
                        /[۰-۹]/g,
                        function (digit) {

                            return String(
                                "۰۱۲۳۴۵۶۷۸۹"
                                .indexOf(digit)
                            );

                        }
                    );

            }


            function normalizePhoneInput(value) {

                let phone =
                    normalizeArabicNumbers(value);

                phone =
                    phone
                    .replace(/[^\d+]/g, "")
                    .replace(
                        /\+/g,
                        function (match, offset) {

                            return offset === 0
                                ? "+"
                                : "";

                        }
                    );


                if (phone.startsWith("+20")) {

                    phone =
                        "0" +
                        phone.slice(3);

                }
                else if (phone.startsWith("0020")) {

                    phone =
                        "0" +
                        phone.slice(4);

                }
                else if (
                    phone.startsWith("20") &&
                    phone.length === 12
                ) {

                    phone =
                        "0" +
                        phone.slice(2);

                }
                else if (
                    /^\d{10}$/.test(phone)
                ) {

                    phone =
                        "0" +
                        phone;

                }


                return phone;

            }


            function cleanName(value) {

                return String(value || "")
                    .replace(/\s+/g, " ")
                    .trim();

            }

            function getErrorMessage(error) {

                if (!error) {

                    return:
                        "حدث خطأ غير متوقع، حاول مرة أخرى.";

                }


                const code =
                    String(
                        error.code || ""
                    ).toLowerCase();


                if (
                    code.includes("network") ||
                    code.includes("offline")
                ) {

                    return:
                        "تعذر الاتصال بالإنترنت. تأكد من اتصالك وحاول مرة أخرى.";

                }


                if (
                    code.includes("permission") ||
                    code.includes("permission-denied")
                ) {

                    return:
                        "لا يمكن إكمال التسجيل حاليًا. حاول مرة أخرى.";

                }


                if (
                    code.includes("already") ||
                    code.includes("exists") ||
                    code.includes("duplicate")
                ) {

                    return:
                        "يوجد حساب مسجل بالفعل بهذا الرقم.";

                }


                if (
                    code.includes("invalid")
                ) {

                    return:
                        "تأكد من صحة البيانات التي أدخلتها.";

                }


                if (error.message) {

                    return:
                        error.message;

                }


                return:
                    "تعذر إنشاء الحساب. حاول مرة أخرى.";

            }

            function updatePasswordStrength() {

                const password =
                    String(
                        passwordInput.value || ""
                    );


                if (!password) {

                    strength.classList.remove(
                        "show"
                    );

                    strengthBar.style.width =
                        "0%";

                    return;

                }


                strength.classList.add(
                    "show"
                );


                let score = 0;


                if (password.length >= 4) {
                    score++;
                }

                if (password.length >= 8) {
                    score++;
                }

                if (/[A-Z]/.test(password)) {
                    score++;
                }

                if (/[a-z]/.test(password)) {
                    score++;
                }

                if (/\d/.test(password)) {
                    score++;
                }

                if (/[^A-Za-z0-9]/.test(password)) {
                    score++;
                }


                const percentage =
                    Math.min(
                        100,
                        Math.round(
                            (score / 6) * 100
                        )
                    );


                strengthBar.style.width =
                    percentage + "%";


                if (score <= 2) {

                    strengthText.textContent =
                        "كلمة المرور ضعيفة";

                    strengthBar.style.background =
                        "#ff5f6d";

                }
                else if (score <= 4) {

                    strengthText.textContent =
                        "كلمة المرور متوسطة";

                    strengthBar.style.background =
                        "#ffbd4a";

                }
                else {

                    strengthText.textContent =
                        "كلمة المرور قوية";

                    strengthBar.style.background =
                        "#31d07c";

                }

            }

            /* =====================================================

            async function waitForModules() {

                const timeout =
                    15000;

                const started =
                    Date.now();


                while (
                    Date.now() - started <
                    timeout
                ) {

                    if (
                        window.FirebaseAuthReady &&
                        window.FirebaseDatabaseReady
                    ) {

                        return true;

                    }


                    await new Promise(
                        function (resolve) {

                            setTimeout(
                                resolve,
                                100
                            );

                        }
                    );

                }


                return !!(
                    window.FirebaseAuth &&
                    window.FirebaseDatabase
                );

            }

            form.addEventListener(

                async function (event) {

                    event.preventDefault();

                    hideMessage();

                    const grade =

                    /* ---------------------------------------------

                        nameInput.focus();

                        return;

                    if (name.length < 3) {

                        );

                        nameInput.focus();

                        return;

                    }


                    /* ---------------------------------------------
                       GRADE
                       --------------------------------------------- */

                        );

                        gradeInput.focus();

                        return;

                    }


                    /* ---------------------------------------------
                       PHONE
                       --------------------------------------------- */

                    if (!phone) {

                        showMessage(
                            "من فضلك اكتب رقم الهاتف."
                        );

                        phoneInput.focus();

                        return;

                    }


                    if (
                        !/^01[0125]\d{8}$/.test(phone)
                    ) {

                        showMessage(
                            "اكتب رقم هاتف مصري صحيح مثل 01012345678."
                        );

                        phoneInput.focus();

                        return;

                    }


                    /* ---------------------------------------------
                       EMAIL
                       --------------------------------------------- */

                    if (email) {

                        const emailPattern =
                            /^[^\s@]+@[^\s@]+\.[^\s@]+$/;


                        if (
                            !emailPattern.test(email)
                        ) {

                            showMessage(
                                "اكتب بريدًا إلكترونيًا صحيحًا أو اتركه فارغًا."
                            );

                            emailInput.focus();

                            return;

                        }

                    }


                    /* ---------------------------------------------
                       PASSWORD
                       --------------------------------------------- */

                    if (!password) {

                        showMessage(
                            "من فضلك اختر PIN أو كلمة مرور."
                        );

                        passwordInput.focus();

                        return;

                    }


                    if (password.length < 4) {

                        showMessage(
                            "PIN / كلمة المرور يجب أن تكون 4 أحرف أو أرقام على الأقل."
                        );

                        passwordInput.focus();

                        return;

                    }

                        showMessage(
                            "كلمة المرور طويلة جدًا."
                        );

                        passwordInput.focus();

                        return;

                    }


                    /* ---------------------------------------------
                       CONFIRM
                       --------------------------------------------- */

                    if (!confirmPassword) {

                        showMessage(
                            "من فضلك أكد كلمة المرور."
                        );

                        confirmPasswordInput.focus();

                        return;

                    }


                    if (
                        password !==
                        confirmPassword
                    ) {

                        showMessage(
                            "كلمتا المرور غير متطابقتين."
                        );

                        confirmPasswordInput.focus();

                        return;

                    }


                    /* ---------------------------------------------
                       TERMS
                       --------------------------------------------- */

                    if (!termsInput.checked) {

                        showMessage(
                            "يجب الموافقة على التعليمات لإكمال التسجيل."
                        );

                        termsInput.focus();

                        return;

                    }


                    /* ---------------------------------------------
                       LOADING
                       --------------------------------------------- */

                    setLoading(true);


                    try {

                        const ready =
                            await waitForModules();


                        if (!ready) {

                            throw new Error(
                                "تعذر تحميل نظام التسجيل."
                            );

                        }


                        const auth =
                            window.FirebaseAuth;


                        if (
                            !auth ||
                            typeof auth.registerStudent !==
                            "function"
                        ) {

                            throw new Error(
                                "نظام التسجيل غير جاهز."
                            );

                        }


                        /* -----------------------------------------
                           REGISTER
                           ----------------------------------------- */

                        const result =
                            await auth.registerStudent({

                                name:
                                    name,

                                grade:
                                    grade,

                                phone:
                                    phone,

                                pin:
                                    password,

                                email:
                                    email || ""

                            });


                        /* -----------------------------------------
                           SUCCESS
                           ----------------------------------------- */

                        const message =
                            (
                                result &&
                                result.message
                            ) ||
                            "تم إنشاء الحساب وإرساله للمراجعة بنجاح.";


                        showMessage(
                            message,
                            "success"
                        );


                        showToast(
                            "تم إنشاء الحساب بنجاح"
                        );


                        /*
                         * الحساب الجديد يكون pending
                         * ولا يتم تسجيل دخوله تلقائيًا.
                         */


                        setTimeout(
                            function () {

                                window.location.replace(
                                    "login.html"
                                );

                            },
                            1800
                        );

                    catch (error) {

                        console.error(
                            "Registration error:",
                            error
                        );


                        showMessage(
                            getErrorMessage(
                                error
                            ),
                            "error"
                        );


                        setLoading(false);

                    }

                }
            );
