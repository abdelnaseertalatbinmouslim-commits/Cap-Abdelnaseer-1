            value !== null &&
            value !== undefined &&
            safeString(value) !== ""
          ) {
            return value;
          }
        }

        return "";
      }

      function hideLoader() {
        loader?.classList.add("hidden");
      }

      function toast(message, type = "success") {
        if (!toastContainer) return;

        const item =
          document.createElement("div");

        item.className =
          `toast ${type}`;

        item.textContent = message;

        toastContainer.appendChild(item);

        setTimeout(() => {

          item.style.opacity = "0";
          item.style.transform =
            "translateY(10px)";

          setTimeout(() => {
            item.remove();
          }, 250);

        }, 3200);
      }

      function getInitials(name) {
        const value =
          safeString(name);

        if (!value) {
          return "؟";
        }

        const parts =
          value
            .split(/\s+/)
            .filter(Boolean);

        if (parts.length === 1) {
          return parts[0].slice(0, 2);
        }

        return (
          (parts[0][0] || "") +
          (parts[1][0] || "")
        );
      }

      async function waitForFirebase() {
        const start =
          Date.now();

        const timeout =
          12000;

        while (
          Date.now() - start <
          timeout
        ) {

          const ready =
            window.FirebaseCoreReady &&
            window.FirebaseAuthReady &&
            window.FirebaseDatabaseReady &&
            window.FirebaseContentReady &&
            window.FirebaseFeaturesReady;

          if (ready) {
            return true;
          }

          await new Promise(
            resolve =>
              setTimeout(resolve, 100)
          );
        }

        return false;
      }

      async function getCurrentStudent() {

              typeof window.FirebaseAuth.getCurrentUser ===
                "function"
            ) {

                await window.FirebaseAuth.getCurrentUser();
            }

          } catch (error) {
            console.warn(
              "Auth current user failed:",
              error
            );
          }
        }

        if (!student) {

          try {

            const raw =
              localStorage.getItem(
                "currentUser"
              );

            if (raw) {
              student =
                JSON.parse(raw);
            }

          } catch (_) {}
        }

        if (!student) {

          try {

            const raw =
              localStorage.getItem(
                "student"
              );

            if (raw) {
              student =
                JSON.parse(raw);
            }

          } catch (_) {}

      function normalizeTerm(value) {

        const raw =
          safeString(value)
            .toLowerCase();

        if (!raw) {
          return "";
        }

        if (
          raw.includes("first") ||
          raw.includes("term 1") ||
          raw.includes("الاول") ||
          raw.includes("الأول")
        ) {
          return "1";
        }

        if (
          raw.includes("second") ||
          raw.includes("term 2") ||
          raw.includes("الثاني")
        ) {
          return "2";
        }

        if (
          raw.includes("third") ||
          raw.includes("term 3") ||
          raw.includes("الثالث")
        ) {
          return "3";
        }

        if (
          raw.includes("fourth") ||
          raw.includes("term 4") ||
          raw.includes("الرابع")
        ) {
          return "4";
        }

        const match =
          raw.match(/[1-4]/);

        return match
          ? match[0]
          : raw;
      }

      function normalizeSummary(item, key = "") {

        const source =
          item &&
          typeof item === "object"
            ? item
            : {};

        const id =
          firstValue(
            source.id,
            source.summaryId,
            source.fileId,
            source.contentId,
            key
          );

        const title =
          firstValue(
            source.title,
            source.name,
            source.summaryTitle,
            source.fileName,
            "ملخص"
          );

        const description =
          firstValue(
            source.description,
            source.details,
            source.shortDescription,
            source.about,
            "ملخص دراسي للمراجعة."
          );

        const subject =
          firstValue(
            source.subject,
            source.subjectName,
            source.course,
            source.courseName,
            ""
          );

        const subjectId =
          firstValue(
            source.subjectId,
            source.courseId,
            source.subjectKey,
            ""
          );

        const term =
          normalizeTerm(
            firstValue(
              source.term,
              source.semester,
              source.termNumber,
              source.studyTerm
            )
          );

        const grade =
          firstValue(
            source.grade,
            source.year,
            source.level,
            source.studyYear,
            ""
          );

        const url =
          firstValue(
            source.pdfUrl,
            source.fileUrl,
            source.downloadURL,
            source.downloadUrl,
            source.url,
            source.link
          );

        const type =
          safeString(
            firstValue(
              source.type,
              source.fileType,
              source.contentType,
              "pdf"
            )
          ).toLowerCase();

        const published =
          source.published !== false &&
          source.active !== false &&
          safeString(source.status)
            .toLowerCase() !==
            "hidden";

        const isPaid =
          source.paid === true ||
          source.isPaid === true ||
          safeString(
            source.access
          ).toLowerCase() === "paid" ||
          safeString(
            source.type
          ).toLowerCase() === "paid";

        const createdAt =
          firstValue(
            source.createdAt,
            source.updatedAt,
            source.timestamp,
            source.date,
            0
          );

        return {
          ...source,
          id: safeString(id),
          title: safeString(title),
          description: safeString(description),
          subject: safeString(subject),
          subjectId: safeString(subjectId),
          term,
          grade: safeString(grade),
          url: safeString(url),
          type,
          published,
          isPaid,
          createdAt
        };
      }

      function looksLikeSummary(item) {

        const source =
          item &&
          typeof item === "object"
            ? item
            : {};

        const type =
          safeString(
            firstValue(
              source.type,
              source.fileType,
              source.contentType
            )
          ).toLowerCase();

        const category =
          safeString(
            firstValue(
              source.category,
              source.contentCategory,
              source.kind
            )
          ).toLowerCase();

        const title =
          safeString(
            firstValue(
              source.title,
              source.name
            )
          ).toLowerCase();

        const explicitSummary =
          source.summary === true ||
          source.isSummary === true ||
          source.summaryType === true;

        if (explicitSummary) {
          return true;
        }

        if (
          category.includes("summary") ||
          category.includes("ملخص")
        ) {
          return true;
        }

        if (
          type.includes("summary") ||
          type.includes("ملخص")
        ) {
          return true;
        }

        if (
          title.includes("summary") ||
          title.includes("ملخص") ||
          title.includes("مراجعة")
        ) {
          return true;
        }

        return false;
      }

      function objectToArray(data) {

        if (!data) {
          return [];
        }

        if (Array.isArray(data)) {
          return data.map(
            (item, index) =>
              normalizeSummary(
                item,
                String(index)
              )
          );
        }

        if (
          typeof data === "object"
        ) {

          return Object.entries(data)
            .map(
              ([key, value]) =>
                normalizeSummary(
                  value,
                  key
                )
            );
        }

        return [];
      }

      function dedupe(items) {

        const map =
          new Map();

        for (const item of items) {

          const identity =
            safeString(
              item.id
            ) ||
            [
              item.title,
              item.subject,
              item.term,
              item.url
            ]
              .map(safeString)
              .join("|");

          if (!identity) {
            continue;
          }

          if (!map.has(identity)) {
            map.set(
              identity,
              item
            );
          }
        }

        return Array.from(
          map.values()
        );
      }

      function sortSummaries(items) {

        return items.sort(
          (a, b) => {

            const aDate =
              Number(
                a.createdAt
              ) || 0;

            const bDate =
              Number(
                b.createdAt
              ) || 0;

            if (
              aDate !== bDate
            ) {
              return bDate - aDate;
            }

            return a.title.localeCompare(
              b.title,
              "ar"
            );
          }
        );
      }

      async function loadSummaries() {

        const collected = [];

        try {

          if (
            window.FirebaseFeatures &&
            typeof window.FirebaseFeatures.getFiles ===
              "function"
          ) {

            const data =
              await window.FirebaseFeatures.getFiles();

            const items =
              objectToArray(data)
                .filter(
                  item =>
                    item.published &&
                    looksLikeSummary(
                      item
                    )
                );

            collected.push(
              ...items
            );
          }

        } catch (error) {

          console.warn(
            "Features summaries failed:",
            error
          );

        }

        if (
          collected.length === 0
        ) {

          try {

            if (
              window.FirebaseContent &&
              typeof window.FirebaseContent.getFiles ===
                "function"
            ) {

              const data =
                await window.FirebaseContent.getFiles();

              const items =
                objectToArray(data)
                  .filter(
                    item =>
                      item.published &&
                      looksLikeSummary(
                        item
                      )
                  );

              collected.push(
                ...items
              );
            }

          } catch (error) {

            console.warn(
              "Content summaries failed:",
              error
            );

          }
        }

        if (
          collected.length === 0
        ) {

          try {

            if (
              window.FirebaseDatabase &&
              typeof window.FirebaseDatabase.files?.getAll ===
                "function"
            ) {

              const data =
                await window.FirebaseDatabase.files.getAll();

              const items =
                objectToArray(data)
                  .filter(
                    item =>
                      item.published &&
                      looksLikeSummary(
                        item
                      )
                  );

              collected.push(
                ...items
              );
            }

          } catch (error) {

            console.warn(
              "Database summaries failed:",
              error
            );

          }
        }

        allSummaries =
          sortSummaries(
            dedupe(
              collected
            )
          );

        render();
      }

      function matchesSearch(item) {

        const query =
          safeString(
            searchInput.value
          ).toLowerCase();

        if (!query) {
          return true;
        }

        const haystack =
          [
            item.title,
            item.description,
            item.subject,
            item.grade,
            item.term
          ]
            .map(
              value =>
                safeString(
                  value
                ).toLowerCase()
            )
            .join(" ");

        return haystack.includes(
          query
        );
      }

      function matchesTerm(item) {

        if (
          activeTerm === "all"
        ) {
          return true;
        }

        return (
          normalizeTerm(
            item.term
          ) === activeTerm
        );
      }

      function getFiltered() {

        return allSummaries.filter(
          item =>
            item.published &&
            matchesSearch(item) &&
            matchesTerm(item)
        );
      }

      function formatDate(value) {

        if (!value) {
          return "";
        }

        let date;

        if (
          typeof value === "number"
        ) {

          date =
            new Date(
              value < 10000000000
                ? value * 1000
                : value
            );

        } else {

          date =
            new Date(value);
        }

        if (
          Number.isNaN(
            date.getTime()
          )
        ) {
          return "";
        }

        return date.toLocaleDateString(
          "ar-EG",
          {
            day: "numeric",
            month: "short",
            year: "numeric"
          }
        );
      }

      function termLabel(term) {

        switch (
          normalizeTerm(term)
        ) {

          case "1":
            return "الترم الأول";

          case "2":
            return "الترم الثاني";

          case "3":
            return "الترم الثالث";

          case "4":
            return "الترم الرابع";

          default:
            return "ملخص";
        }
      }

      function escapeHTML(value) {

        return safeString(value)
          .replace(
            /&/g,
            "&amp;"
          )
          .replace(
            /</g,
            "&lt;"
          )
          .replace(
            />/g,
            "&gt;"
          )
          .replace(
            /"/g,
            "&quot;"
          )
          .replace(
            /'/g,
            "&#039;"
          );
      }

      function getOpenUrl(item) {

        if (!item.url) {
          return "";
        }

        return item.url;
      }

                item.createdAt

                  ${escapeHTML(
                    subjectText
                  )}
                </span>

                    termText
                  )}

      async function init() {

        try {

          await waitForFirebase();

          const student =
            await getCurrentStudent();

          if (!student) {

            window.location.href =
              "login.html?redirect=" +
              encodeURIComponent(
                "summaries.html"
              );

            return;
          }

          fillStudent(
            student
          );

          await loadSummaries();

        } catch (error) {

          console.error(
            "Summaries initialization failed:",
            error
          );

          loadingGrid.style.display =
            "none";

          summaryGrid.style.display =
            "none";

          errorState.classList.add(
            "show"
          );

          resultCount.textContent =
            "تعذر تحميل الملخصات.";

          toast(
            "تعذر تحميل الملخصات.",
            "error"
          );

        } finally {

          hideLoader();

        }
      }

      init();

    })();
