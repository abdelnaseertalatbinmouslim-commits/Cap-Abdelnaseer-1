            /* =================================================
               HELPERS
               ================================================= */

            safeString(value, fallback = "") {

                if (
                    value === null ||
                    value === undefined
                ) {
                    return fallback;
                }

                return String(value).trim() || fallback;

            },


            escapeHtml(value) {

                return this.safeString(value)
                    .replace(/&/g, "&amp;")
                    .replace(/</g, "&lt;")
                    .replace(/>/g, "&gt;")
                    .replace(/"/g, "&quot;")
                    .replace(/'/g, "&#039;");

            },


            firstValue(...values) {

                for (const value of values) {

                    if (
                        value !== null &&
                        value !== undefined &&
                        String(value).trim() !== ""
                    ) {

                        return value;

                    }

                }

                return null;

            },


            toArray(value) {

                if (Array.isArray(value)) {
                    return value;
                }

                if (
                    value &&
                    typeof value === "object"
                ) {

                    return Object.keys(value)
                        .map(key => {

                            const item =
                                value[key];

                            if (
                                item &&
                                typeof item === "object"
                            ) {

                                return {
                                    ...item,
                                    _key:
                                        item._key ||
                                        key
                                };

                            }

                            return {
                                value: item,
                                _key: key
                            };

                        });

                }

                return [];

            },


            getTimestamp(value) {

                if (
                    value === null ||
                    value === undefined ||
                    value === ""
                ) {
                    return null;
                }


                if (
                    typeof value === "object" &&
                    value.seconds
                ) {

                    return Number(
                        value.seconds
                    ) * 1000;

                }


                if (
                    typeof value === "object" &&
                    value.timestamp
                ) {

                    return this.getTimestamp(
                        value.timestamp
                    );

                }


                if (
                    typeof value === "number"
                ) {

                    if (
                        value < 10000000000
                    ) {

                        return value * 1000;

                    }

                    return value;

                }


                const number =
                    Number(value);


                if (
                    !Number.isNaN(number)
                ) {

                    if (
                        number < 10000000000
                    ) {

                        return number * 1000;

                    }

                    return number;

                }


                const parsed =
                    Date.parse(
                        String(value)
                    );


                return Number.isNaN(parsed)
                    ? null
                    : parsed;

            },


            formatDate(value) {

                const timestamp =
                    this.getTimestamp(value);


                if (!timestamp) {
                    return "غير مسجل";
                }


                try {

                    return new Intl.DateTimeFormat(
                        "ar-EG",
                        {
                            year: "numeric",
                            month: "short",
                            day: "numeric",
                            hour: "numeric",
                            minute: "2-digit"
                        }
                    ).format(
                        new Date(timestamp)
                    );

                } catch (error) {

                    return new Date(timestamp)
                        .toLocaleString("ar-EG");

                }

            },


            normalizeBoolean(value) {

                if (
                    value === true ||
                    value === false
                ) {
                    return value;
                }


                const text =
                    this.safeString(
                        value
                    ).toLowerCase();


                if (
                    [
                        "true",
                        "1",
                        "yes",
                        "read",
                        "مقروء",
                        "مقروءة"
                    ].includes(text)
                ) {

                    return true;

                }


                if (
                    [
                        "false",
                        "0",
                        "no",
                        "unread",
                        "غير مقروء",
                        "غير مقروءة"
                    ].includes(text)
                ) {

                    return false;

                }


                return false;

            },

            /* =================================================

               ================================================= */

            async loadReadStates(
                notifications
            ) {

                this.state.readMap = {};


                const DB =
                    window.FirebaseDatabase;


                if (
                    !DB ||
                    !DB.notifications
                ) {

                    return;

                }


                /*
                 * First use the database helper
                 * when available.
                 */

                if (
                    typeof DB.notifications
                        .isRead ===
                    "function"
                ) {

                    for (
                        const notification
                        of notifications
                    ) {

                        if (!notification._id) {
                            continue;
                        }


                        try {

                            const read =
                                await DB
                                    .notifications
                                    .isRead(
                                        this.state.studentKey,
                                        notification._id
                                    );


                            this.state.readMap[
                                notification._id
                            ] =
                                Boolean(read);

                        } catch (error) {

                            console.warn(
                                "Read state failed:",
                                notification._id,
                                error
                            );

                        }

                    }

                    return;

                }


                /*
                 * Direct fallback to the known
                 * legacy structure:
                 *
                 * /notificationReads/{studentKey}
                 */

                try {

                    if (
                        typeof DB.getChild ===
                        "function"
                    ) {

                        const data =
                            await DB.getChild(
                                "notificationReads",
                                this.state.studentKey
                            );


                        if (
                            data &&
                            typeof data === "object"
                        ) {

                            Object.keys(data)
                                .forEach(key => {

                                    const value =
                                        data[key];


                                    this.state.readMap[
                                        key
                                    ] =
                                        Boolean(
                                            value &&
                                            (
                                                value.read === true ||
                                                value === true
                                            )
                                        );

                                });

                        }

                    }

                } catch (error) {

                    console.warn(
                        "Direct read states failed:",
                        error
                    );

                }

            },


            /* =================================================
               LOAD NOTIFICATIONS
               ================================================= */

            async loadNotifications() {

                const DB =
                    window.FirebaseDatabase;


                const Features =
                    window.FirebaseFeatures;


                const collected = [];


                /*
                 * Features first.
                 */

                if (
                    Features &&
                    typeof Features
                        .getNotifications ===
                    "function"
                ) {

                    try {

                        const data =
                            await Features
                                .getNotifications();


                        this.toArray(data)
                            .forEach(item => {

                                const normalized =
                                    this
                                        .normalizeNotification(
                                            item
                                        );


                                if (normalized) {

                                    collected.push(
                                        normalized
                                    );

                                }

                            });

                    } catch (error) {

                        console.warn(
                            "Features notifications failed:",
                            error
                        );

                    }

                }


                /*
                 * Direct DB helper.
                 */

                if (
                    DB &&
                    DB.notifications &&
                    typeof DB.notifications
                        .getAll ===
                    "function"
                ) {

                    try {

                        let data = null;


                        try {

                            data =
                                await DB
                                    .notifications
                                    .getAll(
                                        this.state.student
                                    );

                        } catch (firstError) {

                            try {

                                data =
                                    await DB
                                        .notifications
                                        .getAll();

                            } catch (secondError) {

                                console.warn(
                                    "DB notifications failed:",
                                    secondError
                                );

                            }

                        }


                        this.toArray(data)
                            .forEach(item => {

                                const normalized =
                                    this
                                        .normalizeNotification(
                                            item
                                        );


                                if (normalized) {

                                    collected.push(
                                        normalized
                                    );

                                }

                            });

                    } catch (error) {

                        console.warn(
                            "Notifications loading failed:",
                            error
                        );

                    }

                }


                /*
                 * Remove duplicates.
                 */

                const map =
                    new Map();


                collected.forEach(item => {

                    const key =
                        item._id ||
                        [
                            item._title,
                            item._date,
                            item._message
                        ].join("|");


                    if (!map.has(key)) {

                        map.set(
                            key,
                            item
                        );

                    }

                });


                const result =
                    Array.from(
                        map.values()
                    );


                result.sort(
                    (a, b) =>
                        (b._date || 0) -
                        (a._date || 0)
                );


                return result;

            },


            /* =================================================
               TARGET SAFETY
               ================================================= */

            isForCurrentStudent(notification) {

                if (!notification) {
                    return false;
                }


                const student =
                    this.state.student;


                if (!student) {
                    return true;
                }


                /*
                 * The database layer normally already
                 * filters audience/target/student.
                 *
                 * We only perform extra filtering when
                 * the notification explicitly contains
                 * a student identifier.
                 */

                const explicitStudent =
                    this.firstValue(
                        notification.studentId,
                        notification.studentKey,
                        notification.uid,
                        notification.userId,
                        notification.phone,
                        notification.studentPhone
                    );


                if (!explicitStudent) {

                    return true;

                }


                const currentCandidates = [

                    student._key,

                    student.key,

                    student.studentKey,

                    student.id,

                    student.studentId,

                    student.uid,

                    student.userId,

                    student.phone,

                    student.email

                ]
                    .filter(Boolean)
                    .map(
                        value =>
                            String(value)
                                .trim()
                                .toLowerCase()
                    );


                return currentCandidates.includes(
                    String(explicitStudent)
                        .trim()
                        .toLowerCase()
                );

            },

            setFilter(filter) {

                this.state.filter =

            setSearch(value) {

                this.state.search =
                    this.safeString(
                        value
                    ).toLowerCase();


                this.applyFilters();

            },


            applyFilters() {

                let notifications =
                    [...this.state.allNotifications];


                if (
                    this.state.filter ===
                    "unread"
                ) {

                    notifications =
                        notifications.filter(
                            notification =>
                                !this.isRead(
                                    notification
                                )
                        );

                }


                if (
                    this.state.filter ===
                    "read"
                ) {

                    notifications =
                        notifications.filter(
                            notification =>
                                this.isRead(
                                    notification
                                )
                        );

                }


                const search =
                    this.state.search;


                if (search) {

                    notifications =
                        notifications.filter(
                            notification => {

                                const text = [

                                    notification._title,

                                    notification._message,

                                    notification._type

                                ]
                                    .join(" ")
                                    .toLowerCase();


                                return text.includes(
                                    search
                                );

                            }
                        );

                }


                this.state.filteredNotifications =
                    notifications;


                this.render();

            },


            /* =================================================
               READ CHECK
               ================================================= */

            isRead(notification) {

                if (!notification) {
                    return false;
                }


                const id =
                    notification._id;


                if (
                    id &&
                    Object.prototype.hasOwnProperty.call(
                        this.state.readMap,
                        id
                    )
                ) {

                    return Boolean(
                        this.state.readMap[id]
                    );

                }


                /*
                 * Support notification objects that
                 * already carry read information.
                 */

                if (
                    notification.read === true ||
                    notification.isRead === true
                ) {

                    return true;

                }


                if (
                    notification.read === false ||
                    notification.isRead === false
                ) {

                    return false;

                }


                return false;

            },


            /* =================================================
               UNREAD COUNT
               ================================================= */

            updateUnreadCount() {

                const unread =
                    this.state.allNotifications
                        .filter(
                            notification =>
                                !this.isRead(
                                    notification
                                )
                        )
                        .length;


                const element =
                    document.getElementById(
                        "unreadCount"
                    );


                if (element) {

                    element.textContent =
                        unread;

                }

            },


            /* =================================================
               MARK AS READ
               ================================================= */

            async markAsRead(notification) {

                if (!notification) {
                    return;
                }


                const id =
                    notification._id;


                if (!id) {

                    this.showToast(
                        "الإشعار ده مش مرتبط بمعرف صالح.",
                        "error"
                    );

                    return;

                }


                if (
                    this.isRead(
                        notification
                    )
                ) {

                    return;

                }


                const DB =
                    window.FirebaseDatabase;


                const Features =
                    window.FirebaseFeatures;


                try {

                    let success = false;


                    if (
                        Features &&
                        typeof Features
                            .markNotificationRead ===
                        "function"
                    ) {

                        try {

                            await Features
                                .markNotificationRead(
                                    id
                                );

                            success = true;

                        } catch (error) {

                            console.warn(
                                "Feature mark read failed:",
                                error
                            );

                        }

                    }


                    if (
                        !success &&
                        DB &&
                        DB.notifications &&
                        typeof DB.notifications
                            .markRead ===
                        "function"
                    ) {

                        await DB
                            .notifications
                            .markRead(
                                this.state.studentKey,
                                id
                            );

                        success = true;

                    }


                    if (!success) {

                        throw new Error(
                            "لا توجد دالة متاحة لتسجيل الإشعار كمقروء."
                        );

                    }


                    this.state.readMap[id] =
                        true;


                    this.updateUnreadCount();


                    this.applyFilters();


                } catch (error) {

                    console.error(
                        "Mark notification read failed:",
                        error
                    );


                    this.showToast(
                        "تعذر تسجيل الإشعار كمقروء.",
                        "error"
                    );

                }

            },


            /* =================================================
               RENDER
               ================================================= */

            render() {

                const list =
                    document.getElementById(
                        "notificationsList"
                    );


                if (!list) {
                    return;
                }


                const notifications =
                    this.state.filteredNotifications;


                if (
                    !this.state.allNotifications.length
                ) {

                    this.showOnly(
                        "emptyState"
                    );

                    return;

                }


                if (!notifications.length) {

                    list.innerHTML = `

                        <div
                            class="empty-state"
                            style="width:100%;"
                        >

                            <div class="state-icon">
                                🔎
                            </div>

                            <h2 class="state-title">
                                مفيش إشعارات مطابقة
                            </h2>

                            <p class="state-text">
                                جرّب تغيّر الفلتر أو كلمة البحث.
                            </p>

                        </div>

                    `;


                    this.showOnly(
                        "notificationsList"
                    );


                    return;

                }


                list.innerHTML =
                    notifications
                        .map(
                            notification =>
                                this.renderNotification(
                                    notification
                                )
                        )
                        .join("");


                this.showOnly(
                    "notificationsList"
                );


                list
                    .querySelectorAll(
                        "[data-notification-id]"
                    )
                    .forEach(button => {

                        button.addEventListener(
                            "click",
                            async () => {

                                const id =
                                    button.dataset
                                        .notificationId;


                                const notification =
                                    this
                                        .state
                                        .allNotifications
                                        .find(
                                            item =>
                                                String(
                                                    item._id
                                                ) ===
                                                String(id)
                                        );


                                await this.markAsRead(
                                    notification
                                );

                            }
                        );

                    });

            },


            renderNotification(notification) {

                const read =
                    this.isRead(
                        notification
                    );


                const title =
                    this.escapeHtml(
                        notification._title
                    );


                const message =
                    this.escapeHtml(
                        notification._message
                    );


                const type =
                    this.escapeHtml(
                        notification._type
                    );


                const date =
                    this.escapeHtml(
                        this.formatDate(
                            notification._createdAt
                        )
                    );


                const id =
                    this.escapeHtml(
                        notification._id
                    );


                const icon =
                    this.getNotificationIcon(
                        notification
                    );


                const readButton =
                    read
                        ? ""
                        : `

                            <button
                                type="button"
                                class="read-button"
                                data-notification-id="${id}"
                            >
                                ✓ تحديد كمقروء
                            </button>

                        `;


                return `

                    <article
                        class="
                            notification-card
                            ${read ? "" : "unread"}
                        "
                    >

                        <div class="notification-icon">
                            ${icon}
                        </div>


                        <div class="notification-content">

                            <div class="notification-head">

                                <h2
                                    class="notification-title"
                                >
                                    ${title}
                                </h2>

                                <time
                                    class="notification-date"
                                >
                                    ${date}
                                </time>

                            </div>


                            ${
                                message
                                    ? `
                                        <div
                                            class="notification-message"
                                        >
                                            ${message}
                                        </div>
                                      `
                                    : ""
                            }


                            <div class="notification-meta">

                                <span class="meta-badge">
                                    ${type}
                                </span>

                                ${
                                    read
                                        ? `
                                            <span class="meta-badge">
                                                مقروء
                                            </span>
                                          `
                                        : `
                                            <span
                                                class="meta-badge"
                                                style="
                                                    color:#9db0ff;
                                                    background:rgba(91,124,255,.09);
                                                "
                                            >
                                                جديد
                                            </span>
                                          `
                                }

                            </div>


                            ${readButton}

                        </div>

                    </article>

                `;

            },


            getNotificationIcon(notification) {

                const type =
                    this.safeString(
                        this.firstValue(
                            notification.type,
                            notification.category,
                            notification.notificationType
                        ),
                        ""
                    ).toLowerCase();


                if (
                    type.includes("quiz") ||
                    type.includes("exam") ||
                    type.includes("اختبار")
                ) {

                    return "📝";

                }


                if (
                    type.includes("file") ||
                    type.includes("pdf") ||
                    type.includes("ملف")
                ) {

                    return "📄";

                }


                if (
                    type.includes("video") ||
                    type.includes("فيديو")
                ) {

                    return "🎥";

                }


                if (
                    type.includes("course") ||
                    type.includes("كورس") ||
                    type.includes("مقرر")
                ) {

                    return "📚";

                }


                if (
                    type.includes("warning") ||
                    type.includes("alert") ||
                    type.includes("تحذير")
                ) {

                    return "⚠️";

                }


                if (
                    type.includes("success") ||
                    type.includes("نجاح")
                ) {

                    return "✅";

                }


                return "🔔";

            },


            /* =================================================
               ERROR
               ================================================= */

            getReadableError(error) {

                if (!error) {

                    return "تعذر تحميل الإشعارات.";

                }


                const code =
                    this.safeString(
                        error.code
                    ).toLowerCase();


                if (
                    code.includes("permission")
                ) {

                    return "لا توجد صلاحية كافية لقراءة الإشعارات حاليًا.";

                }


                if (
                    code.includes("network")
                ) {

                    return "تحقق من اتصال الإنترنت وحاول مرة أخرى.";

                }


                if (error.message) {

                    return this.safeString(
                        error.message,
                        "تعذر تحميل الإشعارات."
                    );

                }


                return "تعذر تحميل الإشعارات حاليًا.";

            },


            showError(message) {

                const element =
                    document.getElementById(
                        "errorText"
                    );


                if (element) {

                    element.textContent =
                        message;

                }


                this.showOnly(
                    "errorState"
                );

            },


            /* =================================================
               LOAD PAGE
               ================================================= */

            async load() {

                if (this.state.loading) {
                    return;
                }


                this.state.loading = true;


                this.showLoader();


                try {

                    this.state.student =
                        await this.getCurrentStudent();


                    if (!this.state.student) {

                        window.location.replace(
                            "login.html?redirect=" +
                            encodeURIComponent(
                                "notifications.html"
                            )
                        );

                        return;

                    }


                    this.state.studentKey =
                        this.getStudentKey(
                            this.state.student
                        );


                    this.updateStudentUI();


                    let notifications =
                        await this
                            .loadNotifications();


                    /*
                     * Extra safety:
                     * explicitly addressed notifications
                     * must belong to this student.
                     */

                    notifications =
                        notifications.filter(
                            notification =>
                                this.isForCurrentStudent(
                                    notification
                                )
                        );


                    this.state.allNotifications =
                        notifications;


                    await this.loadReadStates(
                        notifications
                    );


                    this.updateUnreadCount();


                    this.applyFilters();


                    this.hideLoader();


                } catch (error) {

                    console.error(
                        "Notifications page error:",
                        error
                    );


                    this.showError(
                        this.getReadableError(
                            error
                        )
                    );


                    this.hideLoader();

                } finally {

                    this.state.loading = false;

                }

            },


            /* =================================================
               FIREBASE READY
               ================================================= */

            async waitForFirebase(
                timeout = 15000
            ) {

                const start =
                    Date.now();


                while (
                    Date.now() - start <
                    timeout
                ) {

                    if (
                        window.FirebaseCoreReady &&
                        window.FirebaseAuthReady &&
                        window.FirebaseDatabaseReady
                    ) {

                        return true;

                    }


                    await new Promise(
                        resolve =>
                            setTimeout(
                                resolve,
                                80
                            )
                    );

                }


                return Boolean(
                    window.FirebaseDatabase &&
                    window.FirebaseAuth
                );

            },


            /* =================================================
               EVENTS
               ================================================= */

            bindEvents() {

                document
                    .querySelectorAll(
                        ".filter-button"
                    )
                    .forEach(button => {

                        button.addEventListener(
                            "click",
                            () => {

                                this.setFilter(
                                    button.dataset.filter
                                );

                            }
                        );

                    });


                const searchInput =
                    document.getElementById(
                        "searchInput"
                    );


                if (searchInput) {

                    searchInput.addEventListener(
                        "input",
                        event => {

                            this.setSearch(
                                event.target.value
                            );

                        }
                    );

                }


                const retryButton =
                    document.getElementById(
                        "retryButton"
                    );


                if (retryButton) {

                    retryButton.addEventListener(
                        "click",
                        async () => {

                            this.showOnly(
                                "skeletonList"
                            );

                            await this.load();

                        }
                    );

                }


                const mobileButton =
                    document.getElementById(
                        "mobileMenuButton"
                    );


                const mobileNav =
                    document.getElementById(
                        "mobileNav"
                    );


                if (
                    mobileButton &&
                    mobileNav
                ) {

                    mobileButton.addEventListener(
                        "click",
                        () => {

                            mobileNav.classList.toggle(
                                "open"
                            );

                        }
                    );


                    mobileNav
                        .querySelectorAll("a")
                        .forEach(link => {

                            link.addEventListener(
                                "click",
                                () => {

                                    mobileNav
                                        .classList
                                        .remove(
                                            "open"
                                        );

                                }
                            );

                        });

                }


                document.addEventListener(
                    "click",
                    event => {

                        if (
                            !mobileNav ||
                            !mobileButton
                        ) {
                            return;
                        }


                        if (
                            !mobileNav.contains(
                                event.target
                            ) &&
                            !mobileButton.contains(
                                event.target
                            )
                        ) {

                            mobileNav.classList.remove(
                                "open"
                            );

                        }

                    }
                );

            },


            /* =================================================
               INIT
               ================================================= */

            async init() {

                const year =
                    document.getElementById(
                        "currentYear"
                    );


                if (year) {

                    year.textContent =
                        new Date()
                            .getFullYear();

                }


                this.bindEvents();


                try {

                    await this.waitForFirebase();


                    await this.load();

                } catch (error) {

                    console.error(
                        "Notifications initialization failed:",
                        error
                    );


                    this.showError(
                        this.getReadableError(
                            error
                        )
                    );


                    this.hideLoader();

                }

            }

        };


        /* =====================================================
           PUBLIC API
           ===================================================== */

        window.CapNotifications =
            NotificationsPage;


        /* =====================================================
           START
           ===================================================== */

        document.addEventListener(
            "DOMContentLoaded",
            () => {

                NotificationsPage.init();

            }
        );

    </script>
