            function safeString(value) {

                if (
                    value === null ||
                    value === undefined
                ) {
                    return "";
                }

                return String(value).trim();

            }


            function escapeHtml(value) {

                return safeString(value)
                    .replace(/&/g, "&amp;")
                    .replace(/</g, "&lt;")
                    .replace(/>/g, "&gt;")
                    .replace(/"/g, "&quot;")
                    .replace(/'/g, "&#039;");

            }


            function normalizeTerm(value) {

                if (
                    value === null ||
                    value === undefined ||
                    value === ""
                ) {
                    return "";
                }

                const text = safeString(value)
                    .toLowerCase()
                    .replace(
                        /الترم|الفصل|semester|term/g,
                        ""
                    )
                    .trim();

                const arabicNumbers = {
                    "الأول": "1",
                    "الاول": "1",
                    "الأولى": "1",
                    "الاولى": "1",
                    "one": "1",

                    "الثاني": "2",
                    "الثانى": "2",
                    "الثانية": "2",
                    "الثانيه": "2",
                    "two": "2",

                    "الثالث": "3",
                    "الثالثة": "3",
                    "three": "3",

                    "الرابع": "4",
                    "الرابعة": "4",
                    "four": "4"
                };

                if (arabicNumbers[text]) {
                    return arabicNumbers[text];
                }

                const numberMatch =
                    text.match(/[1-4]/);

                if (numberMatch) {
                    return numberMatch[0];
                }

                return text;

            }


            function getSubjectId(subject) {

                return safeString(
                    subject?.id ||
                    subject?.subjectId ||
                    subject?.key ||
                    subject?.code
                );

            }


            function getSubjectName(subject) {

                return safeString(
                    subject?.name ||
                    subject?.title ||
                    subject?.subjectName ||
                    subject?.label ||
                    "مادة دراسية"
                );

            }


            function getSubjectDescription(subject) {

                return safeString(
                    subject?.description ||
                    subject?.desc ||
                    subject?.details ||
                    "استعرض محتوى المادة والملفات والفيديوهات والاختبارات."
                );

            }


            function isPaidSubject(subject) {

                const value = safeString(
                    subject?.paid ??
                    subject?.isPaid ??
                    subject?.premium ??
                    subject?.requiredPayment ??
                    ""
                ).toLowerCase();

                if (
                    value === "true" ||
                    value === "yes" ||
                    value === "paid" ||
                    value === "premium" ||
                    value === "مدفوع"
                ) {
                    return true;
                }

                if (
                    value === "false" ||
                    value === "no" ||
                    value === "free" ||
                    value === "مجاني"
                ) {
                    return false;
                }

                return false;

            }


            function getInitials(name) {

                const text = safeString(name);

                if (!text) {
                    return "ط";
                }

                const words = text
                    .split(/\s+/)
                    .filter(Boolean);

                if (words.length === 1) {
                    return words[0].charAt(0);
                }

                return (
                    words[0].charAt(0) +
                    words[1].charAt(0)
                );

            }

                };

                );

            }

            /* =====================================================
               GET SUBJECTS
            ====================================================== */

            async function loadSubjects() {

                let subjects = [];


                /*
                 * First choice:
                 * FirebaseContent
                 */

                try {

                    if (
                        window.FirebaseContent &&
                        typeof window.FirebaseContent
                            .getSubjects === "function"
                    ) {

                        subjects =
                            await window.FirebaseContent
                                .getSubjects();

                    }

                } catch (error) {

                    console.warn(
                        "FirebaseContent.getSubjects failed:",
                        error
                    );

                }


                /*
                 * Second choice:
                 * FirebaseFeatures
                 */

                if (
                    !Array.isArray(subjects) ||
                    subjects.length === 0
                ) {

                    try {

                        if (
                            window.FirebaseFeatures &&
                            typeof window.FirebaseFeatures
                                .getCourses === "function"
                        ) {

                            const result =
                                await window.FirebaseFeatures
                                    .getCourses();

                            if (
                                Array.isArray(result)
                            ) {
                                subjects = result;
                            } else if (
                                Array.isArray(
                                    result?.subjects
                                )
                            ) {
                                subjects =
                                    result.subjects;
                            }

                        }

                    } catch (error) {

                        console.warn(
                            "FirebaseFeatures.getCourses failed:",
                            error
                        );

                    }

                }


                /*
                 * Third choice:
                 * Direct database subjects.
                 */

                if (
                    !Array.isArray(subjects) ||
                    subjects.length === 0
                ) {

                    try {

                        if (
                            window.FirebaseDatabase &&
                            typeof window.FirebaseDatabase
                                .get === "function"
                        ) {

                            const raw =
                                await window.FirebaseDatabase
                                    .get("subjects");

                            if (
                                raw &&
                                typeof raw === "object"
                            ) {

                                subjects =
                                    Object.keys(raw)
                                        .map(function (key) {

                                            return {
                                                id: key,
                                                ...(
                                                    raw[key] || {}
                                                )
                                            };

                                        });

                            }

                        }

                    } catch (error) {

                        console.warn(
                            "Direct subjects read failed:",
                            error
                        );

                    }

                }


                /*
                 * Normalize result.
                 */

                if (
                    !Array.isArray(subjects)
                ) {
                    subjects = [];
                }


                subjects = subjects
                    .filter(function (subject) {

                        return (
                            subject &&
                            getSubjectId(subject)
                        );

                    })
                    .map(function (subject) {

                        return {
                            ...subject,

                            id:
                                getSubjectId(
                                    subject
                                ),

                            name:
                                getSubjectName(
                                    subject
                                ),

                            description:
                                getSubjectDescription(
                                    subject
                                ),

                            term:
                                normalizeTerm(
                                    subject?.term ||
                                    subject?.semester ||
                                    subject?.year ||
                                    subject?.level ||
                                    ""
                                ),

                            paid:
                                isPaidSubject(
                                    subject
                                )

                        };

                    });


                /*
                 * Remove duplicate IDs.
                 */

                const unique =
                    new Map();


                subjects.forEach(function (subject) {

                    const id =
                        getSubjectId(subject);

                    if (!unique.has(id)) {
                        unique.set(id, subject);
                    }

                });


                state.subjects =
                    Array.from(unique.values());


                return state.subjects;

            }


            /* =====================================================
               FILTERING
            ====================================================== */

            function applyFilters() {

                let result =
                    [...state.subjects];


                /*
                 * Term filter
                 */

                if (
                    state.activeTerm !== "all"
                ) {

                    result =
                        result.filter(
                            function (subject) {

                                return (
                                    normalizeTerm(
                                        subject.term
                                    ) ===
                                    state.activeTerm
                                );

                            }
                        );

                }


                /*
                 * Paid / free filter
                 */

                if (
                    state.activeFilter === "paid"
                ) {

                    result =
                        result.filter(
                            function (subject) {

                                return (
                                    subject.paid === true
                                );

                            }
                        );

                } else if (
                    state.activeFilter === "free"
                ) {

                    result =
                        result.filter(
                            function (subject) {

                                return (
                                    subject.paid !== true
                                );

                            }
                        );

                }


                /*
                 * Preserve Firebase / configured order.
                 * If order exists, sort numerically.
                 */

                result.sort(
                    function (a, b) {

                        const aOrder =
                            Number(
                                a.order ??
                                a.sortOrder ??
                                999999
                            );

                        const bOrder =
                            Number(
                                b.order ??
                                b.sortOrder ??
                                999999
                            );

                        if (
                            !Number.isNaN(aOrder) &&
                            !Number.isNaN(bOrder) &&
                            aOrder !== bOrder
                        ) {

                            return (
                                aOrder -
                                bOrder
                            );

                        }

                        return getSubjectName(a)
                            .localeCompare(
                                getSubjectName(b),
                                "ar"
                            );

                    }
                );


                state.filteredSubjects =
                    result;


                renderSubjects();

            }


            /* =====================================================

            function renderSubjects() {

                    !elements.subjectsGrid ||
                    !elements.emptyState
                ) {
                    return;
                }


                elements.subjectsGrid.innerHTML =
                    "";


                if (
                    !state.filteredSubjects.length
                ) {

                    elements.subjectsGrid.style.display =
                        "none";

                    elements.emptyState.classList.add(
                        "show"
                    );

                } else {

                    elements.emptyState.classList.remove(
                        "show"
                    );

                    elements.subjectsGrid.style.display =
                        "grid";


                    state.filteredSubjects
                        .forEach(
                            function (
                                subject,
                                index
                            ) {

                                const card =
                                    createSubjectCard(
                                        subject,
                                        index
                                    );

                                elements.subjectsGrid
                                    .appendChild(
                                        card
                                    );

                            }
                        );

                }


                updateCountText();

            }


            function createSubjectCard(
                subject,
                index
            ) {

                const card =
                    document.createElement("article");


                card.className =
                    "subject-card";


                card.style.animationDelay =
                    Math.min(
                        index * 0.06,
                        0.5
                    ) + "s";


                const id =
                    getSubjectId(subject);


                const name =
                    getSubjectName(subject);


                const description =
                    getSubjectDescription(subject);


                const icon =
                    getSubjectIcon(subject);


                const term =
                    normalizeTerm(
                        subject?.term
                    );


                const paid =
                    subject?.paid === true;


                const order =
                    index + 1;


                const termText =
                    term === "1"
                        ? "الترم الأول"
                        : term === "2"
                            ? "الترم الثاني"
                            : term === "3"
                                ? "الترم الثالث"
                                : term === "4"
                                    ? "الترم الرابع"
                                    : "مادة دراسية";


                card.innerHTML = `

                    <div class="subject-card-top">

                        <div class="subject-icon">
                            ${escapeHtml(icon)}
                        </div>

                        <div class="subject-number">
                            ${order}
                        </div>

                    </div>


                    <div class="subject-body">

                        <h3 class="subject-name">
                            ${escapeHtml(name)}
                        </h3>


                        <p class="subject-description">
                            ${escapeHtml(description)}
                        </p>


                        <div class="subject-meta">

                            <span class="meta-chip">
                                📚 ${escapeHtml(termText)}
                            </span>

                            <span class="meta-chip ${
                                paid ? "paid" : ""
                            }">

                                ${
                                    paid
                                        ? "⭐ مدفوعة"
                                        : "✓ مجانية"
                                }

                            </span>

                        </div>


                        <div class="subject-footer">

                            <a
                                class="open-subject"
                                href="subject.html?id=${encodeURIComponent(id)}"
                                aria-label="فتح مادة ${escapeHtml(name)}"
                            >

                                دخول إلى المادة

                                <span>
                                    ←
                                </span>

                            </a>

                        </div>

                    </div>

                `;


                return card;

            }


            /* =====================================================
               COUNT
            ====================================================== */

            function updateCountText() {

                if (
                    !elements.subjectsCountText
                ) {
                    return;
                }


                const total =
                    state.filteredSubjects.length;


                const all =
                    state.subjects.length;


                if (
                    state.activeFilter === "all" &&
                    state.activeTerm === "all"
                ) {

                    elements.subjectsCountText.textContent =
                        total === 1
                            ? "مادة واحدة متاحة"
                            : `${total} مواد متاحة`;

                    return;

                }


                elements.subjectsCountText.textContent =
                    `${total} من ${all} مادة`;

            }


            /* =====================================================
               FILTER BUTTONS
            ====================================================== */

            function setupFilters() {

                if (!elements.filters) {
                    return;
                }


                elements.filters
                    .querySelectorAll(
                        ".filter-button"
                    )
                    .forEach(
                        function (button) {

                            button.addEventListener(
                                "click",
                                function () {

                                    elements.filters
                                        .querySelectorAll(
                                            ".filter-button"
                                        )
                                        .forEach(
                                            function (
                                                item
                                            ) {

                                                item.classList
                                                    .remove(
                                                        "active"
                                                    );

                                            }
                                        );


                                    button.classList.add(
                                        "active"
                                    );


                                    state.activeFilter =
                                        safeString(
                                            button.dataset
                                                .filter
                                        ) ||
                                        "all";


                                    applyFilters();

                                }
                            );

                        }
                    );

            }


            /* =====================================================
               TERM BUTTONS
            ====================================================== */

            function setupTerms() {

                if (!elements.terms) {
                    return;
                }


                elements.terms
                    .querySelectorAll(
                        ".term-button"
                    )
                    .forEach(
                        function (button) {

                            button.addEventListener(
                                "click",
                                function () {

                                    elements.terms
                                        .querySelectorAll(
                                            ".term-button"
                                        )
                                        .forEach(
                                            function (
                                                item
                                            ) {

                                                item.classList
                                                    .remove(
                                                        "active"
                                                    );

                                            }
                                        );


                                    button.classList.add(
                                        "active"
                                    );


                                    state.activeTerm =
                                        safeString(
                                            button.dataset
                                                .term
                                        ) ||
                                        "all";


                                    applyFilters();

                                }
                            );

                        }
                    );

            }


            /* =====================================================
               DYNAMIC TERMS
            ====================================================== */

            function updateAvailableTerms() {

                if (!elements.terms) {
                    return;
                }


                const available =
                    new Set();


                state.subjects.forEach(
                    function (subject) {

                        const term =
                            normalizeTerm(
                                subject?.term
                            );

                        if (term) {
                            available.add(term);
                        }

                    }
                );


                elements.terms
                    .querySelectorAll(
                        ".term-button"
                    )
                    .forEach(
                        function (button) {

                            const term =
                                safeString(
                                    button.dataset.term
                                );


                            if (
                                term === "all"
                            ) {
                                return;
                            }


                            /*
                             * We keep all four terms visible
                             * so the future subjects can appear
                             * without changing the UI.
                             */

                            if (
                                available.has(term)
                            ) {

                                button.style.opacity =
                                    "1";

                                button.style.pointerEvents =
                                    "auto";

                            } else {

                                button.style.opacity =
                                    "0.55";

                            }

                        }
                    );

            }


            /* =====================================================
               LOAD
            ====================================================== */

            async function initialize() {

                showLoader();


                try {

                    setupMobileMenu();

                    setupFilters();

                    setupTerms();

                    setupAuth();


                    if (
                        elements.currentYear
                    ) {

                        elements.currentYear
                            .textContent =
                            new Date()
                                .getFullYear();

                    }


                    /*
                     * Wait for Firebase modules if the
                     * readiness promises exist.
                     */

                    try {

                        if (
                            window.FirebaseCoreReadyPromise
                        ) {

                            await Promise.resolve(
                                window.FirebaseCoreReadyPromise
                            );

                        }

                    } catch (error) {
                        console.warn(
                            "Firebase core wait failed:",
                            error
                        );
                    }


                    try {

                        if (
                            window.FirebaseAuthReadyPromise
                        ) {

                            await Promise.resolve(
                                window.FirebaseAuthReadyPromise
                            );

                        }

                    } catch (error) {
                        console.warn(
                            "Firebase auth wait failed:",
                            error
                        );
                    }


                    try {

                        if (
                            window.FirebaseDatabaseReadyPromise
                        ) {

                            await Promise.resolve(
                                window.FirebaseDatabaseReadyPromise
                            );

                        }

                    } catch (error) {
                        console.warn(
                            "Firebase database wait failed:",
                            error
                        );
                    }


                    try {

                        if (
                            window.FirebaseContentReadyPromise
                        ) {

                            await Promise.resolve(
                                window.FirebaseContentReadyPromise
                            );

                        }

                    } catch (error) {
                        console.warn(
                            "Firebase content wait failed:",
                            error
                        );
                    }


                    /*
                     * Load subjects.
                     */

                    await loadSubjects();


                    /*
                     * Update available terms.
                     */

                    updateAvailableTerms();


                    /*
                     * Apply initial filters.
                     */

                    applyFilters();


                    state.loading = false;


                    if (
                        state.subjects.length === 0
                    ) {

                        showToast(
                            "لم يتم العثور على مواد منشورة حاليًا.",
                            "error"
                        );

                    }

                } catch (error) {

                    console.error(
                        "Courses initialization error:",
                        error
                    );


                    state.subjects = [];
                    state.filteredSubjects = [];


                    renderSubjects();


                    showToast(
                        "حصلت مشكلة أثناء تحميل المواد. حاول تحديث الصفحة.",
                        "error"
                    );

                } finally {

                    hideLoader();

                }

            }


            /* =====================================================
               PUBLIC API
            ====================================================== */

            window.CapCourses = {

                state: state,

                reload: async function () {

                    await loadSubjects();

                    updateAvailableTerms();

                    applyFilters();

                },

                filterByTerm: function (term) {

                    state.activeTerm =
                        normalizeTerm(term) ||
                        "all";

                    applyFilters();

                },

                filterByType: function (type) {

                    const value =
                        safeString(type)
                            .toLowerCase();

                    if (
                        value === "paid"
                    ) {

                        state.activeFilter =
                            "paid";

                    } else if (
                        value === "free"
                    ) {

                        state.activeFilter =
                            "free";

                    } else {

                        state.activeFilter =
                            "all";

                    }

                    applyFilters();

                },

                getSubjects: function () {

                    return [
                        ...state.subjects
                    ];

                },

                getFilteredSubjects:
                    function () {

                        return [
                            ...state.filteredSubjects
                        ];

                    }

            };


            /* =====================================================
               START
            ====================================================== */

            if (
                document.readyState ===
                "loading"
            ) {

                document.addEventListener(
                    "DOMContentLoaded",
                    initialize
                );

            } else {

                initialize();

            }

        })();
    </script>
