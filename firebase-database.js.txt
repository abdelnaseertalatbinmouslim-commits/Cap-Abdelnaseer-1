  }

  const PATHS = Object.freeze(
    Object.assign(

        students: "students",

        videos: "videos",
        notifications: "notifications",
        notificationReads: "notificationReads",
        quizzes: "quizzes",
        quizAttempts: "quizAttempts",
        results: "results",
        legacyResults: "quiz_results",
        contentViews: "contentViews",
        support: "settings/support",
        supportTickets: "supportTickets",
        settings: "settings",
        stats: "stats",
        passwordResetRequests: "passwordResetRequests"
      }
    )

  /* =========================================================
     BASIC HELPERS
     ========================================================= */

  function isObject(value) {
    return (
      value !== null &&
      typeof value === "object" &&
      !Array.isArray(value)
    );
  }

  function hasValue(value) {
    return value !== undefined && value !== null;
  }

  function safeString(value, fallback) {
    if (value === undefined || value === null) {
      return fallback !== undefined ? String(fallback) : "";
    }

    try {
      return String(value);
    } catch (error) {
      return fallback !== undefined ? String(fallback) : "";
    }
  }

  function cloneObject(value) {
    if (!hasValue(value)) {
      return value;
    }

    try {
      return JSON.parse(JSON.stringify(value));
    } catch (error) {
      return value;
    }
  }

  function normalizeIdentifier(value) {
    return safeString(value, "")
      .trim()
      .toLowerCase();
  }

  function normalizePhone(value) {
    let phone = safeString(value, "").trim();

    if (!phone) {
      return "";
    }

    const arabicDigits = "٠١٢٣٤٥٦٧٨٩";
    const persianDigits = "۰۱۲۳۴۵۶۷۸۹";

    phone = phone.replace(/[٠-٩]/g, function (digit) {
      return String(arabicDigits.indexOf(digit));
    });

    phone = phone.replace(/[۰-۹]/g, function (digit) {
      return String(persianDigits.indexOf(digit));
    });

    phone = phone.replace(/[^\d+]/g, "");

    if (phone.indexOf("00") === 0) {
      phone = "+" + phone.substring(2);
    }

    if (phone.indexOf("+20") === 0) {
      phone = phone.substring(3);
    } else if (phone.indexOf("20") === 0 && phone.length >= 12) {
      phone = phone.substring(2);
    }

    if (phone.charAt(0) === "1" && phone.length === 10) {
      phone = "0" + phone;
    }

    if (phone.length === 10 && phone.indexOf("01") !== 0) {
      phone = "0" + phone;
    }

    return phone;
  }

  function toArray(value) {
    if (Array.isArray(value)) {
      return value.map(function (item, index) {
        if (isObject(item)) {
          return Object.assign(
            {
              _key: String(index),
              key: String(index)
            },
            item
          );
        }

        return {
          _key: String(index),
          key: String(index),
          value: item
        };
      });
    }

    if (isObject(value)) {
      return Object.keys(value).map(function (key) {
        const item = value[key];

        if (isObject(item)) {
          return Object.assign(
            {
              _key: key,
              key: key
            },
            item
          );
        }

        return {
          _key: key,
          key: key,
          value: item
        };
      });
    }

    return [];
  }

  function firstValue() {
    for (let i = 0; i < arguments.length; i++) {
      const value = arguments[i];

      if (
        value !== undefined &&
        value !== null &&
        value !== ""
      ) {
        return value;
      }
    }

    return "";
  }

  function timestampValue() {
    if (
      Core &&
      typeof Core.serverTimestamp === "function"
    ) {
      return Core.serverTimestamp();
    }

    if (
      window.firebase &&
      window.firebase.database &&
      window.firebase.database.ServerValue
    ) {
      return window.firebase.database.ServerValue.TIMESTAMP;
    }

    return Date.now();
  }

  function getErrorMessage(error) {
    if (Core && typeof Core.getErrorMessage === "function") {
      return Core.getErrorMessage(error);
    }

    return safeString(
      error && (error.message || error.code),
      "حدث خطأ غير متوقع."
    );
  }

  function makeError(message, code) {
    const error = new Error(message);

    if (code) {
      error.code = code;
    }

    return error;
  }

  function normalizeStatus(value) {
    return safeString(value, "")
      .trim()
      .toLowerCase();
  }

  function isPublished(value) {
    if (value === true) {
      return true;
    }

    if (value === false) {
      return false;
    }

    const status = normalizeStatus(value);

    if (!status) {
      return true;
    }

    return ![
      "false",
      "0",
      "draft",
      "unpublished",
      "inactive",
      "disabled",
      "مسودة",
      "غير منشور",
      "غير منشورة"
    ].includes(status);
  }

  function numericValue(value, fallback) {
    const number = Number(value);

    return Number.isFinite(number)
      ? number
      : fallback;
  }

  function dateValue(item) {
    if (!item) {
      return 0;
    }

    return numericValue(
      firstValue(
        item.createdAt,
        item.updatedAt,
        item.publishedAt,
        item.date,
        item.timestamp
      ),
      0
    );
  }

  function orderValue(item) {
    if (!item) {
      return 999999;
    }

    return numericValue(
      firstValue(
        item.order,
        item.sortOrder,
        item.position,
        item.index
      ),
      999999
    );
  }

  function sortByOrderAndDate(a, b) {
    const orderDifference =
      orderValue(a) - orderValue(b);

    if (orderDifference !== 0) {
      return orderDifference;
    }

    return dateValue(b) - dateValue(a);
  }

  function sortNewestFirst(a, b) {
    return dateValue(b) - dateValue(a);
  }

  function stripInternalKeys(data) {
    if (!isObject(data)) {
      return data;
    }

    const result = cloneObject(data);

    delete result._key;
    delete result.key;

    return result;
  }

  /* =========================================================
     CURRENT USER / IDENTITY
     ========================================================= */

  function getCurrentUser() {
    if (
      window.FirebaseAuth &&
      typeof window.FirebaseAuth.getCurrentUser === "function"
    ) {
      return window.FirebaseAuth.getCurrentUser();
    }

    if (Core && typeof Core.getCurrentUser === "function") {
      return Core.getCurrentUser();
    }

    return null;
  }

  function getLocalIdentity() {
    const keys = [
      "currentUser",
      "student",
      "currentStudent"
    ];

    const result = {
      raw: null,
      key: "",
      uid: "",
      id: "",
      studentId: "",
      userId: "",
      phone: "",
      email: "",
      name: ""
    };

    for (let i = 0; i < keys.length; i++) {
      try {
        const raw = localStorage.getItem(keys[i]);

        if (!raw) {
          continue;
        }

        const parsed = JSON.parse(raw);

        if (isObject(parsed)) {
          result.raw = parsed;
          break;
        }
      } catch (error) {
        // Ignore malformed local storage.
      }
    }

    try {
      result.key =
        localStorage.getItem("currentStudentKey") || "";
    } catch (error) {
      result.key = "";
    }

    const source = result.raw || {};

    result.uid = safeString(
      firstValue(
        source.uid,
        source.authUid,
        source.firebaseUid
      )
    );

    result.id = safeString(
      firstValue(
        source.id,
        source.studentId,
        source.userId
      )
    );

    result.studentId = safeString(
      source.studentId
    );

    result.userId = safeString(
      source.userId
    );

    result.phone = normalizePhone(
      firstValue(
        source.phone,
        source.mobile,
        source.phoneNumber
      )
    );

    result.email = normalizeIdentifier(
      source.email
    );

    result.name = safeString(
      firstValue(
        source.name,
        source.studentName,
        source.fullName
      )
    );

    return result;
  }

  function getIdentityCandidates() {
    const candidates = [];

    function addCandidate(type, value) {
      if (!hasValue(value) || value === "") {
        return;
      }

      candidates.push({
        type: type,
        value: safeString(value).trim(),
        normalized:
          type === "phone"
            ? normalizePhone(value)
            : normalizeIdentifier(value)
      });
    }

    const local = getLocalIdentity();

    addCandidate("key", local.key);
    addCandidate("uid", local.uid);
    addCandidate("id", local.id);
    addCandidate("studentId", local.studentId);
    addCandidate("userId", local.userId);
    addCandidate("phone", local.phone);
    addCandidate("email", local.email);

    const firebaseUser = getCurrentUser();

    if (firebaseUser) {
      addCandidate("uid", firebaseUser.uid);
      addCandidate("email", firebaseUser.email);
      addCandidate("phone", firebaseUser.phoneNumber);
    }

    return candidates;
  }

  function studentStatusRank(student) {
    const status = normalizeStatus(
      student && student.status
    );

    if (
      [
        "approved",
        "active",
        "مقبول",
        "نشط"
      ].includes(status)
    ) {
      return 3;
    }

    if (
      [
        "pending",
        "waiting",
        "قيد المراجعة",
        "قيد الانتظار"
      ].includes(status)
    ) {
      return 2;
    }

    if (
      [
        "rejected",
        "blocked",
        "مرفوض",
        "محظور"
      ].includes(status)
    ) {
      return 1;
    }

    return 0;
  }

  function studentMatchesCandidate(student, key, candidate) {
    if (!isObject(student) || !candidate) {
      return false;
    }

    const type = candidate.type;
    const value = candidate.normalized;

    if (!value) {
      return false;
    }

    if (type === "key") {
      return (
        normalizeIdentifier(key) === value
      );
    }

    if (type === "uid") {
      return (
        normalizeIdentifier(student.uid) === value ||
        normalizeIdentifier(student.authUid) === value ||
        normalizeIdentifier(student.firebaseUid) === value
      );
    }

    if (
      type === "id" ||
      type === "studentId" ||
      type === "userId"
    ) {
      return (
        normalizeIdentifier(student.id) === value ||
        normalizeIdentifier(student.studentId) === value ||
        normalizeIdentifier(student.userId) === value
      );
    }

    if (type === "phone") {
      return (
        normalizePhone(student.phone) === value ||
        normalizePhone(student.mobile) === value ||
        normalizePhone(student.phoneNumber) === value
      );
    }

    if (type === "email") {
      return (
        normalizeIdentifier(student.email) === value
      );
    }

    return false;
  }

  function chooseBestStudent(matches) {
    if (!matches || !matches.length) {
      return null;
    }

    return matches.sort(function (a, b) {
      const aExact =
        a.exactKey ? 1 : 0;

      const bExact =
        b.exactKey ? 1 : 0;

      if (aExact !== bExact) {
        return bExact - aExact;
      }

      const statusDifference =
        studentStatusRank(b.student) -
        studentStatusRank(a.student);

      if (statusDifference !== 0) {
        return statusDifference;
      }

      return (
        dateValue(b.student) -
        dateValue(a.student)
      );
    })[0];
  }

  /* =========================================================
     GENERIC DATABASE OPERATIONS
     ========================================================= */

  async function ready() {
    return true;
  }

  function ref(path) {
    if (!path) {
      return db.ref();
    }

    return db.ref(String(path));
  }

  async function get(path) {
    try {
      const snapshot = await ref(path).once("value");

      return snapshot.exists()
        ? snapshot.val()
        : null;
    } catch (error) {
      throw error;
    }
  }

  async function exists(path) {
    try {
      const snapshot = await ref(path).once("value");

      return snapshot.exists();
    } catch (error) {
      throw error;
    }
  }

  function listen(path, callback, errorCallback) {
    const databaseRef = ref(path);

    const handler = function (snapshot) {
      if (typeof callback === "function") {
        callback(
          snapshot.exists()
            ? snapshot.val()
            : null,
          snapshot
        );
      }
    };

    const errorHandler = function (error) {
      if (typeof errorCallback === "function") {
        errorCallback(error);
      } else {
        console.error(
          "Firebase listener error:",
          error
        );
      }
    };

    databaseRef.on(
      "value",
      handler,
      errorHandler
    );

    return function unsubscribe() {
      databaseRef.off(
        "value",
        handler
      );
    };
  }

  async function update(path, data) {
    if (!path) {
      throw makeError(
        "مسار Firebase غير صالح.",
        "invalid-path"
      );
    }

    if (!isObject(data)) {
      throw makeError(
        "بيانات التحديث غير صالحة.",
        "invalid-data"
      );
    }

    return ref(path).update(data);
  }

  async function set(path, data) {
    if (!path) {
      throw makeError(
        "مسار Firebase غير صالح.",
        "invalid-path"
      );
    }

    return ref(path).set(data);
  }

  async function push(path, data) {
    if (!path) {
      throw makeError(
        "مسار Firebase غير صالح.",
        "invalid-path"
      );
    }

    const childRef = ref(path).push();

    if (data === undefined) {
      await childRef.set(null);
    } else {
      await childRef.set(data);
    }

    return {
      key: childRef.key,
      ref: childRef
    };
  }

  function section(path) {
    return ref(path);
  }

  async function getStudentsRaw() {
    return get(PATHS.students);
  }

  async function getStudents() {
    const data = await getStudentsRaw();

    return toArray(data);
  }

  async function getStudentByKey(key) {
    if (!key) {
      return null;
    }

    const data = await get(
      PATHS.students + "/" + key
    );

    if (!isObject(data)) {
      return null;
    }

    return Object.assign(
      {
        _key: String(key),
        key: String(key)
      },
      data
    );
  }

  async function findStudent() {
    const data = await getStudentsRaw();

    if (!data) {
      return null;
    }

    const candidates =
      getIdentityCandidates();

    if (!candidates.length) {
      return null;
    }

    const students = toArray(data);
    const matches = [];

    students.forEach(function (student) {
      const key = firstValue(
        student._key,
        student.key
      );

      let matched = false;
      let exactKey = false;

      for (let i = 0; i < candidates.length; i++) {
        if (
          studentMatchesCandidate(
            student,
            key,
            candidates[i]
          )
        ) {
          matched = true;

          if (
            candidates[i].type === "key" &&
            normalizeIdentifier(key) ===
              candidates[i].normalized
          ) {
            exactKey = true;
          }

          break;
        }
      }

      if (matched) {
        matches.push({
          student: student,
          exactKey: exactKey
        });
      }
    });

    const selected =
      chooseBestStudent(matches);

    return selected
      ? selected.student
      : null;
  }

  async function findStudentByPhone(phone) {
    const normalized =
      normalizePhone(phone);

    if (!normalized) {
      return null;
    }

    const data = await getStudentsRaw();

    if (!data) {
      return null;
    }

    const matches = [];

    toArray(data).forEach(function (student) {
      const studentPhone =
        normalizePhone(
          firstValue(
            student.phone,
            student.mobile,
            student.phoneNumber
          )
        );

      if (
        studentPhone &&
        studentPhone === normalized
      ) {
        matches.push({
          student: student,
          exactKey: false
        });
      }
    });

    const selected =
      chooseBestStudent(matches);

    return selected
      ? selected.student
      : null;
  }

  async function getCurrentStudent() {
    return findStudent();
  }

  async function linkStudentToAuth(
    student,
    user
  ) {
    if (!student || !user) {
      return student;
    }

    const key = firstValue(
      student._key,
      student.key
    );

    if (!key) {
      return student;
    }

    const updates = {};

    if (
      user.uid &&
      !student.uid &&
      !student.authUid &&
      !student.firebaseUid
    ) {
      updates.uid = user.uid;
    }

    if (
      user.email &&
      !student.email
    ) {
      updates.email = user.email;
    }

    if (
      user.phoneNumber &&
      !student.phone &&
      !student.mobile &&
      !student.phoneNumber
    ) {
      updates.phone =
        normalizePhone(user.phoneNumber);
    }

    if (!Object.keys(updates).length) {
      return student;
    }

    try {
      await update(
        PATHS.students + "/" + key,
        updates
      );

      return Object.assign(
        {},
        student,
        updates
      );
    } catch (error) {
      /*
       * Linking is intentionally non-blocking.
       * Existing student data must remain usable
       * even if this optional update is denied.
       */
      return student;
    }
  }

  async function updateStudent(
    studentKey,
    data
  ) {
    if (!studentKey) {
      throw makeError(
        "معرّف الطالب غير موجود.",
        "invalid-student-key"
      );
    }

    if (!isObject(data)) {
      throw makeError(
        "بيانات الطالب غير صالحة.",
        "invalid-student-data"
      );
    }

    const safeData =
      stripInternalKeys(data);

    return update(
      PATHS.students + "/" + studentKey,
      safeData
    );
  }

  function normalizeFile(item) {
    if (!isObject(item)) {
      return null;
    }

    const result = cloneObject(item);

    result._key = firstValue(
      item._key,
      item.key,
      item.id,
      ""
    );

    result.key = result._key;

    result.id = firstValue(
      item.id,
      result._key
    );

    result.title = firstValue(
      item.title,
      item.name,
      item.fileName,
      item.filename,
      "ملف"
    );

    result.name = firstValue(
      item.name,
      item.title,
      result.title
    );

    result.url = firstValue(
      item.pdfUrl,
      item.fileUrl,
      item.url,
      item.link,
      item.downloadURL,
      item.downloadUrl
    );

    result.subject = firstValue(
      item.subject,
      item.subjectId,
      item.course,
      ""
    );

    result.term = firstValue(
      item.term,
      item.semester,
      ""
    );

    result.grade = firstValue(
      item.grade,
      item.year,
      item.level,
      ""
    );

    return result;
  }

  function matchesFilterValue(
    itemValue,
    filterValue
  ) {
    if (
      filterValue === undefined ||
      filterValue === null ||
      filterValue === ""
    ) {
      return true;
    }

    if (
      itemValue === undefined ||
      itemValue === null
    ) {
      return false;
    }

    return (
      normalizeIdentifier(itemValue) ===
        normalizeIdentifier(filterValue) ||
      safeString(itemValue) ===
        safeString(filterValue)
    );
  }

  async function getFiles(filters) {
    const data = await get(PATHS.files);

    let files = toArray(data)
      .map(normalizeFile)
      .filter(Boolean);

    filters = filters || {};

    files = files.filter(function (file) {
      const published =
        firstValue(
          file.published,
          file.active,
          file.isPublished,
          file.status
        );

      if (
        filters.published !== false &&
        published !== undefined &&
        published !== null &&
        !isPublished(published)
      ) {
        return false;
      }

      if (
        !matchesFilterValue(
          file.subject,
          firstValue(
            filters.subject,
            filters.subjectId,
            filters.course
          )
        )
      ) {
        return false;
      }

      if (
        !matchesFilterValue(
          file.term,
          firstValue(
            filters.term,
            filters.semester
          )
        )
      ) {
        return false;
      }

      if (
        !matchesFilterValue(
          file.grade,
          firstValue(
            filters.grade,
            filters.year,
            filters.level
          )
        )
      ) {
        return false;
      }

      return true;
    });

    files.sort(sortByOrderAndDate);

    return files;
  }

  function getVideoUrlType(url) {
    const value = safeString(url, "")
      .trim()
      .toLowerCase();

    if (!value) {
      return "unknown";
    }

    if (
      value.indexOf("youtube.com") !== -1 ||
      value.indexOf("youtu.be") !== -1
    ) {
      return "youtube";
    }

    if (
      value.indexOf("vimeo.com") !== -1
    ) {
      return "vimeo";
    }

    if (
      value.indexOf("<iframe") !== -1
    ) {
      return "iframe";
    }

    return "direct";
  }

  async function getVideos(filters) {
    const data = await get(PATHS.videos);

    let videos = toArray(data)
      .map(normalizeVideo)
      .filter(Boolean);

    filters = filters || {};

    videos = videos.filter(function (video) {
      const published =
        firstValue(
          video.published,
          video.active,
          video.isPublished,
          video.status
        );

     ========================================================= */

  function resultMatchesStudent(
    result,
    key,
    student
  ) {
    if (!isObject(result)) {
      return false;
    }

    const candidates = [];

    if (student) {
      candidates.push(
        {
          type: "key",
          value: firstValue(
            student._key,
            student.key
          )
        },
        {
          type: "uid",
          value: firstValue(
            student.uid,
            student.authUid
          )
        },
        {
          type: "id",
          value: firstValue(
            student.id,
            student.studentId
          )
        },
        {
          type: "phone",
          value: firstValue(
            student.phone,
            student.mobile,
            student.phoneNumber
          )
        },
        {
          type: "email",
          value: student.email
        }
      );
    }

    candidates.push({
      type: "key",
      value: key
    });

    return candidates.some(
      function (candidate) {
        const candidateValue =
          candidate.type === "phone"
            ? normalizePhone(
                candidate.value
              )
            : normalizeIdentifier(
                candidate.value
              );

        if (!candidateValue) {
          return false;
        }

        if (candidate.type === "key") {
          return (
            normalizeIdentifier(
              key
            ) === candidateValue ||
            normalizeIdentifier(
              result.studentKey
            ) === candidateValue ||
            normalizeIdentifier(
              result.studentId
            ) === candidateValue ||
            normalizeIdentifier(
              result.userId
            ) === candidateValue
          );
        }

        if (candidate.type === "uid") {
          return (
            normalizeIdentifier(
              result.uid
            ) === candidateValue ||
            normalizeIdentifier(
              result.userId
            ) === candidateValue
          );
        }

        if (candidate.type === "id") {
          return (
            normalizeIdentifier(
              result.id
            ) === candidateValue ||
            normalizeIdentifier(
              result.studentId
            ) === candidateValue ||
            normalizeIdentifier(
              result.userId
            ) === candidateValue
          );
        }

        if (candidate.type === "phone") {
          return (
            normalizePhone(
              result.phone
            ) === candidateValue ||
            normalizePhone(
              result.mobile
            ) === candidateValue ||
            normalizePhone(
              result.phoneNumber
            ) === candidateValue
          );
        }

        if (candidate.type === "email") {
          return (
            normalizeIdentifier(
              result.email
            ) === candidateValue
          );
        }

        return false;
      }
    );
  }

     NOTIFICATIONS
     ========================================================= */

  function notificationMatchesStudent(
    notification,
    student
  ) {
    if (!isObject(notification)) {
      return false;
    }

    if (!student) {
      return true;
    }

    const target =
      normalizeIdentifier(
        firstValue(
          notification.target,
          notification.audience,
          notification.targetType,
          "all"
        )
      );

    if (
      [
        "all",
        "everyone",
        "public",
        "all_students",
        "الكل",
        "الجميع"
      ].includes(target)
    ) {
      return true;
    }

    const studentKey =
      firstValue(
        student._key,
        student.key
      );

    const studentId =
      firstValue(
        student.id,
        student.studentId,
        student.userId
      );

    const studentPhone =
      normalizePhone(
        firstValue(
          student.phone,
          student.mobile,
          student.phoneNumber
        )
      );

    const notificationStudent =
      firstValue(
        notification.studentId,
        notification.studentKey,
        notification.userId,
        notification.uid,
        notification.phone
      );

    if (notificationStudent) {
      const normalizedTarget =
        notification.phone
          ? normalizePhone(
              notification.phone
            )
          : normalizeIdentifier(
              notificationStudent
            );

      if (
        normalizedTarget ===
          normalizeIdentifier(studentKey) ||
        normalizedTarget ===
          normalizeIdentifier(studentId) ||
        normalizedTarget ===
          normalizeIdentifier(student.uid) ||
        normalizedTarget ===
          normalizeIdentifier(student.email) ||
        normalizedTarget ===
          studentPhone
      ) {
        return true;
      }

      return false;
    }

    const notificationGrade =
      firstValue(
        notification.grade,
        notification.year,
        notification.level
      );

    if (
      notificationGrade &&
      student.grade
    ) {
      return (
        normalizeIdentifier(
          notificationGrade
        ) ===
        normalizeIdentifier(
          student.grade
        )
      );
    }

    if (
      notification.grade ||
      notification.year ||
      notification.level
    ) {
      return false;
    }

    return true;
  }

    student
  ) {

      await getNotifications(student);

      student &&
      firstValue(
        student._key,

      );

        await isNotificationRead(
          studentKey,
          firstValue(
            notification.id,
            notification._key
          )

      if (!read) {
        result.push(notification);
      }
    }

    return result;

      await get(PATHS.stats);

    return isObject(data)
      ? data
      : {};
  }

  /* =========================================================
     PASSWORD RESET REQUESTS
     ========================================================= */

  async function createPasswordResetRequest(
    data
  ) {
    if (!isObject(data)) {
      throw makeError(
        "بيانات طلب إعادة تعيين كلمة المرور غير صالحة.",
        "invalid-reset-request"
      );
    }

    const payload =
      stripInternalKeys(data);

    payload.createdAt =
      firstValue(
        payload.createdAt,
        timestampValue()
      );

    payload.status =
      firstValue(
        payload.status,
        "pending"
      );

    return push(
      PATHS.passwordResetRequests,
      payload
    );
  }

  /* =========================================================
     GENERIC CHILD HELPERS
     ========================================================= */

  async function getChild(
    path,
    childPath
  ) {
    const fullPath =
      String(path).replace(/\/+$/, "") +
      "/" +
      String(childPath).replace(/^\/+/, "");

    return get(fullPath);
  }

  async function updateChild(
    path,
    childPath,
    data
  ) {
    const fullPath =
      String(path).replace(/\/+$/, "") +
      "/" +
      String(childPath).replace(/^\/+/, "");

    return update(fullPath, data);
  }
