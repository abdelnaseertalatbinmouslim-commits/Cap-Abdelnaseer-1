            function safeString(value) {

                if (
                    value === null ||
                    value === undefined
                ) {
                    return "";
                }

                return String(value).trim();
            }


            function normalizeText(value) {

                return safeString(value)
                    .toLowerCase()
                    .replace(/[أإآ]/g, "ا")
                    .replace(/ة/g, "ه")
                    .replace(/ى/g, "ي")
                    .replace(/\s+/g, " ")
                    .trim();
            }


            function normalizeId(value) {

                return safeString(value)
                    .toLowerCase()
                    .replace(/\s+/g, "-");
            }


            function escapeHtml(value) {

                return safeString(value)
                    .replace(/&/g, "&amp;")
                    .replace(/</g, "&lt;")
                    .replace(/>/g, "&gt;")
                    .replace(/"/g, "&quot;")
                    .replace(/'/g, "&#039;");
            }


            function firstValue() {

                for (
                    let i = 0;
                    i < arguments.length;
                    i++
                ) {

                    const value =
                        arguments[i];

                    if (
                        value !== undefined &&
                        value !== null &&
                        safeString(value) !== ""
                    ) {

                        return value;
                    }

                }

                return "";
            }


            function toArray(value) {

                if (Array.isArray(value)) {
                    return value;
                }


                if (
                    value &&
                    typeof value === "object"
                ) {

                    return Object.keys(value)
                        .map(function (key) {

                            const item =
                                value[key];

                            if (
                                item &&
                                typeof item === "object"
                            ) {

                                return Object.assign(
                                    {},
                                    item,
                                    {
                                        key:
                                            item.key ||
                                            key
                                    }
                                );

                            }

                            return item;

                        });
                }


                return [];
            }


            function timestampValue(value) {

                if (
                    value === null ||
                    value === undefined ||
                    value === ""
                ) {
                    return 0;
                }


                if (
                    typeof value === "number" &&
                    isFinite(value)
                ) {
                    return value;
                }


                if (
                    value &&
                    typeof value === "object"
                ) {

                    if (
                        typeof value.toMillis ===
                        "function"
                    ) {

                        return value.toMillis();
                    }


                    if (
                        typeof value.seconds ===
                        "number"
                    ) {

                        return value.seconds * 1000;
                    }


                    if (
                        typeof value.timestamp ===
                        "number"
                    ) {

                        return value.timestamp;
                    }
                }


                const parsed =
                    Date.parse(
                        String(value)
                    );


                return isNaN(parsed)
                    ? 0
                    : parsed;
            }


            function formatDate(value) {

                const timestamp =
                    timestampValue(value);


                if (!timestamp) {
                    return "";
                }


                try {

                    return new Intl.DateTimeFormat(
                        "ar-EG",
                        {
                            day: "numeric",
                            month: "short",
                            year: "numeric"
                        }
                    ).format(
                        new Date(timestamp)
                    );

                } catch (error) {

                    return "";
                }
            }


            function normalizeTerm(value) {

                const text =
                    normalizeText(value);


                if (!text) {
                    return "";
                }


                if (
                    text === "1" ||
                    text.includes("اول")
                ) {
                    return "1";
                }


                if (

                ) {
                    return "3";
                }


                if (
                    text === "4" ||
                    text.includes("رابع")
                ) {
                    return "4";
                }


                return safeString(value);
            }


            function isPublished(item) {

                if (!item) {
                    return true;
                }


                const value =
                    firstValue(
                        item.published,
                        item.isPublished,
                        item.active,
                        item.enabled,
                        item.visible,
                        item.status
                    );


                if (
                    value === "" ||
                    value === undefined ||
                    value === null
                ) {
                    return true;
                }


                if (
                    typeof value === "boolean"
                ) {
                    return value;
                }


                const text =
                    normalizeText(value);


                if (
                    [
                        "false",
                        "0",
                        "no",
                        "draft",
                        "hidden",
                        "inactive",
                        "disabled",
                        "unpublished",
                        "مخفي",
                        "غير منشور"
                    ].includes(text)
                ) {

                    return false;
                }


                return true;
            }

            function getVideoUrl(video) {

                if (!video) {

                }


                return safeString(

            }


            function detectVideoType(url) {

                const value =
                    safeString(url)
                        .toLowerCase();


                if (!value) {
                    return "unknown";
                }


                if (
                    value.includes("youtube.com") ||
                    value.includes("youtu.be")
                ) {

                    return "youtube";
                }


                if (
                    value.includes("vimeo.com")
                ) {

                    return "vimeo";
                }


                if (
                    value.includes("<iframe") ||
                    value.includes("iframe")
                ) {

                    return "iframe";
                }


                if (
                    /\.(mp4|webm|ogg)(\?|#|$)/i
                        .test(value)
                ) {

                    return "direct";
                }


                return "external";
            }

                    ) {

                        id =
                            parsed.pathname
                                .replace("/", "")
                                .split("/")[0];

                    } else {

                        id =
                            parsed.searchParams.get(
                                "v"
                            ) || "";

                        if (
                            !id &&
                            parsed.pathname.includes(
                                "/embed/"
                            )
                        ) {

                            id =
                                parsed.pathname
                                    .split("/embed/")[1]
                                    .split("/")[0];

                        }

                    }


                    if (!id) {
                        return "";
                    }


                    return (
                        "https://www.youtube.com/embed/" +
                        encodeURIComponent(id) +
                        "?rel=0"
                    );

                } catch (error) {

                    return "";
                }
            }


            function getVimeoEmbedUrl(url) {

                const value =
                    safeString(url);


                const match =
                    value.match(
                        /vimeo\.com\/(?:video\/)?(\d+)/i
                    );


                if (!match) {
                    return "";
                }


                return (
                    "https://player.vimeo.com/video/" +
                    encodeURIComponent(match[1])
                );
            }


            function isProbablyIframe(value) {

                const text =
                    safeString(value);


                return (
                    text.includes("<iframe") ||
                    text.includes("iframe")
                );
            }


            /* =====================================================
               NORMALIZE VIDEO
               ===================================================== */

            function normalizeVideo(
                video,
                index
            ) {

                const item =
                    video &&
                    typeof video === "object"
                        ? video
                        : {};


                const id =
                    safeString(
                        firstValue(
                            item.id,
                            item.videoId,
                            item.key,
                            item.uid
                        )
                    ) ||
                    "video-" + index;


                const title =
                    safeString(
                        firstValue(
                            item.title,
                            item.name,
                            item.videoTitle,
                            item.label
                        )
                    ) ||
                    "فيديو تعليمي";


                const description =
                    safeString(
                        firstValue(
                            item.description,
                            item.desc,
                            item.details,
                            item.note
                        )
                    );


                const subjectId =
                    normalizeId(
                        firstValue(
                            item.subjectId,
                            item.subject,
                            item.courseId,
                            item.course
                        )
                    );


                const subjectName =
                    safeString(
                        firstValue(
                            item.subjectName,
                            item.subjectTitle,
                            typeof item.subject ===
                            "object"
                                ? item.subject.name
                                : "",
                            item.courseName
                        )
                    );


                const term =
                    normalizeTerm(
                        firstValue(
                            item.term,
                            item.semester,
                            item.termId
                        )
                    );


                const url =
                    getVideoUrl(item);


                const createdAt =
                    firstValue(
                        item.createdAt,
                        item.uploadedAt,
                        item.date,
                        item.timestamp,
                        item.updatedAt
                    );


                const duration =
                    safeString(
                        firstValue(
                            item.duration,
                            item.durationText,
                            item.videoDuration
                        )
                    );


                const order =
                    Number(
                        firstValue(
                            item.order,
                            item.sortOrder,
                            item.position,
                            999999
                        )
                    );


                return Object.assign(
                    {},
                    item,
                    {
                        id: id,

                        title: title,

                        description:
                            description,

                        subjectId:
                            subjectId,

                        subjectName:
                            subjectName,

                        term:
                            term,

                        url:
                            url,

                        urlType:
                            detectVideoType(
                                url
                            ),

                        duration:
                            duration,

                        createdAt:
                            createdAt,

                        order:
                            isFinite(order)
                                ? order
                                : 999999,

                        published:
                            isPublished(item)
                    }
                );
            }

            function normalizeSubject(
                subject,
                index
            ) {

                const item =
                    subject &&

                        ? subject
                        : {};


                const id =
                    normalizeId(
                        firstValue(
                            item.id,
                            item.subjectId,
                            item.key,
                            item.slug
                        )
                    ) ||
                    "subject-" + index;

                    safeString(
                        firstValue(
                            item.name,
                            item.title,
                            item.subjectName,

                        )
                    ) ||
                    id;


                const term =
                    normalizeTerm(
                        firstValue(

                            item.termId
                        )
                    );


                return Object.assign(
                    {},
                    item,

                        name: name,
                        term: term
                    }
                );
            }


            async function loadSubjects() {

                let subjects = [];


                try {

                    if (
                        window.FirebaseContent &&
                        typeof
                        window.FirebaseContent
                            .getSubjects ===
                        "function"
                    ) {

                        subjects =
                            await window.FirebaseContent
                                .getSubjects();

                    }

                } catch (error) {

                    console.warn(
                        "Content subjects failed:",
                        error
                    );
                }


                if (
                    (!subjects ||
                    !subjects.length) &&
                    window.FirebaseDatabase
                ) {

                    try {

                        const raw =
                            await window.FirebaseDatabase
                                .get("subjects");

                        subjects =
                            toArray(raw);

                    } catch (error) {

                        console.warn(
                            "Direct subjects failed:",
                            error
                        );
                    }
                }


                state.subjects =
                    toArray(subjects)
                        .map(
                            normalizeSubject
                        )
                        .filter(
                            function (subject) {

                                return (
                                    subject &&
                                    subject.id &&
                                    subject.name
                                );

                            }
                        );


                const map = {};


                state.subjects.forEach(
                    function (subject) {

                        if (!map[subject.id]) {

                            map[subject.id] =
                                subject;

                        }

                    }
                );


                state.subjects =
                    Object.keys(map)
                        .map(function (key) {

                            return map[key];

                        });


                populateSubjectFilter();
            }


            function populateSubjectFilter() {

                if (!subjectFilter) {
                    return;
                }


                const current =
                    state.subject;


                subjectFilter.innerHTML = `
                    <option value="all">
                        كل المواد
                    </option>
                `;


                state.subjects
                    .slice()
                    .sort(function (a, b) {

                        return a.name.localeCompare(
                            b.name,
                            "ar"
                        );

                    })
                    .forEach(
                        function (subject) {

                            const option =
                                document.createElement(
                                    "option"
                                );


                            option.value =
                                subject.id;


                            option.textContent =
                                subject.name;


                            subjectFilter
                                .appendChild(
                                    option
                                );

                        }
                    );


                if (
                    current &&
                    current !== "all"
                ) {

                    const exists =
                        state.subjects.some(
                            function (subject) {

                                return (
                                    subject.id ===
                                    current
                                );

                            }
                        );


                    if (exists) {

                        subjectFilter.value =
                            current;

                    } else {

                        const option =
                            document.createElement(
                                "option"
                            );


                        option.value =
                            current;


                        option.textContent =
                            state.subjectFromUrl ||
                            current;


                        subjectFilter
                            .appendChild(
                                option
                            );


                        subjectFilter.value =
                            current;
                    }

                } else {

                    subjectFilter.value =
                        "all";
                }


                updateActiveSubject();
            }

                            .getVideos ===
                        "function"
                    ) {

                        videos =
                            await window.FirebaseFeatures
                                .getVideos();

                    /*
                     * DATABASE FALLBACK
                     */

                        window.FirebaseDatabase &&
                        window.FirebaseDatabase.videos &&
                        typeof
                        window.FirebaseDatabase
                            .videos
                            .getAll ===

                    ) {

            function getFriendlyError(error) {

                if (!error) {

                    return (
                        "تعذر تحميل الفيديوهات حاليًا."
                    );
                }


                if (
                    window.FirebaseCore &&
                    typeof
                    window.FirebaseCore
                        .getErrorMessage ===
                    "function"
                ) {

                    try {

                        const message =
                            window.FirebaseCore
                                .getErrorMessage(
                                    error
                                );


                        if (message) {
                            return message;
                        }

                    } catch (ignored) {}
                }


                if (error.message) {
                    return error.message;
                }


                return (
                    "تعذر تحميل الفيديوهات حاليًا."
                );
            }


            /* =====================================================
               FILTERS
               ===================================================== */

            function applyFilters() {

                const search =
                    normalizeText(
                        state.search
                    );


                state.filteredVideos =
                    state.videos.filter(
                        function (video) {

                            if (search) {

                                const searchable =
                                    normalizeText(
                                        [
                                            video.title,
                                            video.description,
                                            video.subjectName,
                                            video.urlType
                                        ].join(" ")
                                    );


                                if (
                                    !searchable.includes(
                                        search
                                    )
                                ) {

                                    return false;
                                }
                            }


                            if (
                                state.term !== "all" &&
                                state.term
                            ) {

                                if (
                                    normalizeTerm(
                                        video.term
                                    ) !==
                                    state.term
                                ) {

                                    return false;
                                }
                            }


                            if (
                                state.subject !==
                                "all" &&
                                state.subject
                            ) {

                                if (
                                    normalizeId(
                                        video.subjectId
                                    ) !==
                                    normalizeId(
                                        state.subject
                                    )
                                ) {

                                    return false;
                                }
                            }


                            return true;

                        }
                    );


                sortVideos();

                renderVideos();

                updateResultsCount();

                updateActiveSubject();

                updatePageText();
            }


            function sortVideos() {

                state.filteredVideos.sort(
                    function (a, b) {

                        const orderA =
                            Number(
                                a.order ||
                                999999
                            );


                        const orderB =
                            Number(
                                b.order ||
                                999999
                            );


                        if (
                            orderA !== orderB
                        ) {

                            return (
                                orderA -
                                orderB
                            );
                        }


                        const dateA =
                            timestampValue(
                                a.createdAt
                            );


                        const dateB =
                            timestampValue(
                                b.createdAt
                            );


                        return (
                            dateB -
                            dateA
                        );

                    }
                );
            }

            function extractYoutubeId(url) {

                const value =
                    safeString(url);


                if (!value) {
                    return "";
                }


                try {

                    const parsed =
                        new URL(value);


                    if (
                        parsed.hostname.includes(
                            "youtu.be"
                        )
                    ) {

                        return (
                            parsed.pathname
                                .replace("/", "")
                                .split("/")[0]
                        );
                    }


                    const queryId =
                        parsed.searchParams.get(
                            "v"
                        );


                    if (queryId) {
                        return queryId;
                    }


                    const match =
                        parsed.pathname.match(
                            /\/(?:embed|shorts)\/([^/?]+)/i
                        );


                    return match
                        ? match[1]
                        : "";

                } catch (error) {

                    return "";
                }
            }


            function getVideoTypeLabel(type) {

                switch (type) {

                    case "youtube":
                        return "YouTube";

                    case "vimeo":
                        return "Vimeo";

                    case "direct":
                        return "فيديو";

                    case "iframe":
                        return "مشغل";

                    case "external":
                        return "رابط فيديو";

                    default:
                        return "فيديو";

                }
            }


            function getSubjectName(id) {

                const normalized =
                    normalizeId(id);


                if (!normalized) {
                    return "";
                }


                const found =
                    state.subjects.find(
                        function (subject) {

                            return (
                                normalizeId(
                                    subject.id
                                ) === normalized
                            );

                        }
                    );


                return found
                    ? found.name
                    : "";
            }

            function renderPlayer(video) {

                if (!playerContainer) {
                    return;
                }


                playerContainer.innerHTML =
                    "";


                const url =
                    safeString(
                        video.url
                    );


                const type =
                    video.urlType ||
                    detectVideoType(
                        url
                    );


                /*
                 * YouTube
                 */

                if (type === "youtube") {

                    const embed =
                        getYoutubeEmbedUrl(
                            url
                        );


                    if (embed) {

                        const iframe =
                            document.createElement(
                                "iframe"
                            );


                        iframe.src =
                            embed;

                        iframe.title =
                            video.title ||
                            "فيديو تعليمي";

                        iframe.allow =
                            "accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share";

                        iframe.allowFullscreen =
                            true;


                        playerContainer
                            .appendChild(
                                iframe
                            );

                        return;
                    }
                }


                /*
                 * Vimeo
                 */

                if (type === "vimeo") {

                    const embed =
                        getVimeoEmbedUrl(
                            url
                        );


                    if (embed) {

                        const iframe =
                            document.createElement(
                                "iframe"
                            );


                        iframe.src =
                            embed;

                        iframe.title =
                            video.title ||
                            "فيديو تعليمي";

                        iframe.allow =
                            "autoplay; fullscreen; picture-in-picture";

                        iframe.allowFullscreen =
                            true;


                        playerContainer
                            .appendChild(
                                iframe
                            );

                        return;
                    }
                }


                /*
                 * iframe code
                 */

                if (
                    type === "iframe" ||
                    isProbablyIframe(url)
                ) {

                    const wrapper =
                        document.createElement(
                            "div"
                        );


                    wrapper.innerHTML =
                        url;


                    const iframe =
                        wrapper.querySelector(
                            "iframe"
                        );


                    if (iframe) {

                        iframe.removeAttribute(
                            "width"
                        );

                        iframe.removeAttribute(
                            "height"
                        );

                        iframe.style.width =
                            "100%";

                        iframe.style.height =
                            "100%";

                        iframe.style.border =
                            "0";


                        playerContainer
                            .appendChild(
                                iframe
                            );

                        return;
                    }
                }


                /*
                 * Direct video
                 */

                if (type === "direct") {

                    const videoElement =
                        document.createElement(
                            "video"
                        );


                    videoElement.className =
                        "direct-video";


                    videoElement.src =
                        url;


                    videoElement.controls =
                        true;


                    videoElement.autoplay =
                        true;


                    videoElement.playsInline =
                        true;


                    videoElement.preload =
                        "metadata";


                    playerContainer
                        .appendChild(
                            videoElement
                        );


                    return;
                }


                /*
                 * External fallback
                 */

                const wrapper =
                    document.createElement(
                        "div"
                    );


                wrapper.style.cssText = `
                    width:100%;
                    height:100%;
                    display:flex;
                    align-items:center;
                    justify-content:center;
                    flex-direction:column;
                    gap:15px;
                    padding:25px;
                    text-align:center;
                    background:#080b12;
                `;


                wrapper.innerHTML = `

                    <div style="
                        font-size:42px;
                    ">
                        🎬
                    </div>

                    <div style="
                        color:#f7f9fc;
                        font-size:15px;
                        font-weight:800;
                    ">
                        الفيديو متاح من خلال رابط خارجي
                    </div>

                    <a
                        href="${escapeHtml(url)}"
                        target="_blank"
                        rel="noopener noreferrer"
                        style="
                            min-height:42px;
                            display:inline-flex;
                            align-items:center;
                            justify-content:center;
                            padding:0 18px;
                            border-radius:10px;
                            background:#6c63ff;
                            color:#fff;
                            font-size:12px;
                            font-weight:800;
                        "
                    >
                        فتح الفيديو
                        ↗
                    </a>
                `;


                playerContainer
                    .appendChild(
                        wrapper
                    );
            }

            async function trackVideoView(video) {

                try {

                    if (
                        window.FirebaseFeatures &&
                        typeof
                        window.FirebaseFeatures
                            .trackView ===
                        "function"
                    ) {

                        await window.FirebaseFeatures
                            .trackView(
                                "video",
                                video.id,
                                {
                                    title:
                                        video.title,

                                    subjectId:
                                        video.subjectId,

                                    term:
                                        video.term
                                }
                            );

                        return;
                    }


                    if (
                        window.FirebaseDatabase &&
                        window.FirebaseDatabase
                            .contentViews &&
                        typeof
                        window.FirebaseDatabase
                            .contentViews
                            .save ===
                        "function"
                    ) {

                        await window.FirebaseDatabase
                            .contentViews
                            .save(
                                "video",
                                video.id,
                                {
                                    title:
                                        video.title,

                                    subjectId:
                                        video.subjectId,

                                    term:
                                        video.term
                                }
                            );
                    }

                } catch (error) {

                    /*
                     * View tracking optional.
                     * لا يمنع تشغيل الفيديو.
                     */

                    console.warn(
                        "Video tracking failed:",
                        error
                    );
                }
            }

            /* =====================================================
               RESET
               ===================================================== */

            function resetFilters() {

                state.search =
                    "";

                state.term =
                    "all";

                state.subject =
                    "all";

                state.subjectFromUrl =
                    "";

                state.termFromUrl =
                    "";


                if (searchInput) {

                    searchInput.value =
                        "";
                }


                if (termFilter) {

                    termFilter.value =
                        "all";
                }


                if (subjectFilter) {

                    subjectFilter.value =
                        "all";
                }


                applyFilters();


                try {

                    const url =
                        new URL(
                            window.location.href
                        );


                    [
                        "subject",
                        "subjectId",
                        "course",
                        "term",
                        "semester"
                    ].forEach(
                        function (key) {

                            url.searchParams.delete(
                                key
                            );

                        }
                    );


                    window.history.replaceState(
                        {},
                        "",
                        url.toString()
                    );

                } catch (error) {

                    console.warn(
                        "URL reset failed:",
                        error
                    );
                }
            }


            function removeSubjectFilterHandler() {

                state.subject =
                    "all";

                state.subjectFromUrl =
                    "";


                if (subjectFilter) {

                    subjectFilter.value =
                        "all";
                }


                try {

                    const url =
                        new URL(
                            window.location.href
                        );


                    url.searchParams.delete(
                        "subject"
                    );

                    url.searchParams.delete(
                        "subjectId"
                    );

                    url.searchParams.delete(
                        "course"
                    );


                    window.history.replaceState(
                        {},
                        "",
                        url.toString()
                    );

                } catch (error) {

                    console.warn(
                        "URL update failed:",
                        error
                    );
                }


                applyFilters();
            }


            /* =====================================================
               URL
               ===================================================== */

            function readUrlParams() {

                const params =
                    new URLSearchParams(
                        window.location.search
                    );


                state.subjectFromUrl =
                    safeString(
                        firstValue(
                            params.get(
                                "subjectId"
                            ),
                            params.get(
                                "subject"
                            ),
                            params.get(
                                "course"
                            )
                        )
                    );


                state.termFromUrl =
                    normalizeTerm(
                        firstValue(
                            params.get(
                                "term"
                            ),
                            params.get(
                                "semester"
                            )
                        )
                    );


                if (
                    state.subjectFromUrl
                ) {

                    state.subject =
                        normalizeId(
                            state.subjectFromUrl
                        );
                }


                if (
                    state.termFromUrl
                ) {

                    state.term =
                        state.termFromUrl;
                }
            }

            async function loadCurrentUser() {

                try {

                    if (
                        window.FirebaseAuth &&
                        typeof
                        window.FirebaseAuth
                            .getCurrentUser ===
                        "function"
                    ) {

                        state.currentUser =
                            window.FirebaseAuth
                                .getCurrentUser();

                    } else if (
                        window.FirebaseDatabase &&
                        typeof
                        window.FirebaseDatabase
                            .getCurrentUser ===
                        "function"
                    ) {

                        state.currentUser =
                            window.FirebaseDatabase
                                .getCurrentUser();
                    }

                } catch (error) {

                    console.warn(
                        "Current user failed:",
                        error
                    );
                }


                updateHeader();
            }

                    );

                        dashboardButton.textContent =
                            "إنشاء حساب";

                        dashboardButton.href =

                    }

            }


            function listenForAuthChanges() {

                try {

                    if (
                        window.FirebaseAuth &&
                        typeof
                        window.FirebaseAuth
                            .onAuthStateChanged ===
                        "function"
                    ) {

                        window.FirebaseAuth
                            .onAuthStateChanged(
                                function (user) {

                                    if (user) {

                                        state.currentUser =
                                            user;
                                    }

                                    updateHeader();

                                }
                            );
                    }

                } catch (error) {

                    console.warn(
                        "Auth listener failed:",
                        error
                    );
                }
            }


            /* =====================================================
               MOBILE MENU
               ===================================================== */

            function closeMobileMenu() {

                if (!mobileNav) {
                    return;
                }


                mobileNav.classList.remove(
                    "open"
                );


                document.body.classList.remove(
                    "menu-open"
                );


                if (mobileMenuButton) {

                    mobileMenuButton
                        .setAttribute(
                            "aria-expanded",
                            "false"
                        );


                    mobileMenuButton.textContent =
                        "☰";
                }
            }


            function toggleMobileMenu() {

                if (!mobileNav) {
                    return;
                }


                const open =
                    mobileNav.classList.toggle(
                        "open"
                    );


                document.body.classList.toggle(
                    "menu-open",
                    open
                );


                if (mobileMenuButton) {

                    mobileMenuButton
                        .setAttribute(
                            "aria-expanded",
                            String(open)
                        );


                    mobileMenuButton.textContent =
                        open
                            ? "×"
                            : "☰";
                }
            }


            /* =====================================================
               EVENTS
               ===================================================== */

            function bindEvents() {

                if (mobileMenuButton) {

                    mobileMenuButton
                        .addEventListener(
                            "click",
                            toggleMobileMenu
                        );
                }


                if (mobileNav) {

                    mobileNav
                        .querySelectorAll("a")
                        .forEach(
                            function (link) {

                                link.addEventListener(
                                    "click",
                                    closeMobileMenu
                                );

                            }
                        );
                }


                if (searchInput) {

                    searchInput
                        .addEventListener(
                            "input",
                            function (event) {

                                state.search =
                                    event.target.value;

                                applyFilters();

                            }
                        );
                }


                if (termFilter) {

                    termFilter
                        .addEventListener(
                            "change",
                            function (event) {

                                state.term =
                                    normalizeTerm(
                                        event.target.value
                                    ) ||
                                    "all";

                                applyFilters();

                            }
                        );
                }


                if (subjectFilter) {

                    subjectFilter
                        .addEventListener(
                            "change",
                            function (event) {

                                state.subject =
                                    normalizeId(
                                        event.target.value
                                    ) ||
                                    "all";


                                try {

                                    const url =
                                        new URL(
                                            window.location.href
                                        );


                                    if (
                                        state.subject ===
                                        "all"
                                    ) {

                                        url.searchParams
                                            .delete(
                                                "subject"
                                            );

                                        url.searchParams
                                            .delete(
                                                "subjectId"
                                            );

                                        url.searchParams
                                            .delete(
                                                "course"
                                            );

                                    } else {

                                        url.searchParams
                                            .set(
                                                "subjectId",
                                                state.subject
                                            );
                                    }


                                    window.history
                                        .replaceState(
                                            {},
                                            "",
                                            url.toString()
                                        );

                                } catch (error) {

                                    console.warn(
                                        "URL filter failed:",
                                        error
                                    );
                                }


                                applyFilters();

                            }
                        );
                }


                if (removeSubjectFilter) {

                    removeSubjectFilter
                        .addEventListener(
                            "click",
                            removeSubjectFilterHandler
                        );
                }


                if (emptyResetButton) {

                    emptyResetButton
                        .addEventListener(
                            "click",
                            resetFilters
                        );
                }


                if (retryButton) {

                    retryButton
                        .addEventListener(
                            "click",
                            loadVideos
                        );
                }


                if (modalClose) {

                    modalClose
                        .addEventListener(
                            "click",
                            closeVideo
                        );
                }


                if (videoModal) {

                    videoModal
                        .addEventListener(
                            "click",
                            function (event) {

                                if (
                                    event.target ===
                                    videoModal
                                ) {

                                    closeVideo();
                                }

                            }
                        );
                }


                document.addEventListener(
                    "keydown",
                    function (event) {

                        if (
                            event.key ===
                            "Escape"
                        ) {

                            if (
                                videoModal &&
                                videoModal.classList
                                    .contains(
                                        "open"
                                    )
                            ) {

                                closeVideo();

                            } else {

                                closeMobileMenu();
                            }
                        }

                    }
                );


                window.addEventListener(
                    "resize",
                    function () {

                        if (
                            window.innerWidth >
                            1050
                        ) {

                            closeMobileMenu();
                        }

                    }
                );
            }


            /* =====================================================
               FIREBASE WAIT
               ===================================================== */

            async function waitForFirebaseModules() {

                const maxWait =
                    15000;


                const started =
                    Date.now();


                while (
                    Date.now() - started <
                    maxWait
                ) {

                    if (
                        window.FirebaseCoreReady &&
                        window.FirebaseAuthReady &&
                        window.FirebaseDatabaseReady &&
                        window.FirebaseContentReady &&
                        window.FirebaseFeaturesReady
                    ) {

                        return true;
                    }


                    if (
                        window.FirebaseDatabase &&
                        window.FirebaseContent &&
                        window.FirebaseFeatures
                    ) {

                        return true;
                    }


                    await new Promise(
                        function (resolve) {

                            setTimeout(
                                resolve,
                                80
                            );

                        }
                    );
                }


                return false;
            }

               INITIALIZE

                if (currentYear) {

                    currentYear.textContent =
                        new Date()
                            .getFullYear();

                readUrlParams();

                await waitForFirebaseModules();

                await loadCurrentUser();

                listenForAuthChanges();


                try {

                    await loadSubjects();

                } catch (error) {

                    console.warn(
                        "Subjects init failed:",
                        error
                    );
                }


                await loadVideos();


                hidePageLoader();
            }
