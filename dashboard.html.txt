    <script>
        (function () {
            "use strict";

            /* =====================================================
               BASIC HELPERS
               ===================================================== */

            const $ = (selector) =>
                document.querySelector(selector);

            const $$ = (selector) =>
                document.querySelectorAll(selector);

            const safeString = (value, fallback = "") => {
                if (
                    value === null ||
                    value === undefined
                ) {
                    return fallback;
                }

                const text = String(value).trim();

                return text || fallback;
            };

            const escapeHTML = (value) => {
                return safeString(value)
                    .replace(/&/g, "&amp;")
                    .replace(/</g, "&lt;")
                    .replace(/>/g, "&gt;")
                    .replace(/"/g, "&quot;")
                    .replace(/'/g, "&#039;");
            };

            const getInitial = (name) => {
                const value = safeString(name, "ط");

                return value.charAt(0);
            };

            const numberValue = (value) => {
                const number = Number(value);

                return Number.isFinite(number)
                    ? number
                    : 0;
            };

            /* =====================================================
               TOAST
               ===================================================== */

            function showToast(
                message,
                type = "info"
            ) {
                const container =
                    $("#toastContainer");

                if (!container) {
                    return;
                }

                const toast =
                    document.createElement("div");

                toast.className = "toast";

                let icon = "i";

                if (type === "success") {
                    icon = "✓";
                }

                if (type === "error") {
                    icon = "!";
                }

                if (type === "warning") {
                    icon = "!";
                }

                toast.innerHTML = `
                    <div class="toast-icon">
                        ${icon}
                    </div>

                    <div class="toast-text">
                        ${escapeHTML(message)}
                    </div>

                    <button
                        type="button"
                        class="toast-close"
                        aria-label="إغلاق"
                    >
                        ×
                    </button>
                `;

                container.appendChild(toast);

                const close =
                    toast.querySelector(
                        ".toast-close"
                    );

                if (close) {
                    close.addEventListener(
                        "click",
                        () => {
                            toast.remove();
                        }
                    );
                }

                setTimeout(() => {
                    if (toast.isConnected) {
                        toast.style.opacity = "0";
                        toast.style.transform =
                            "translateY(10px)";

                        setTimeout(
                            () => toast.remove(),
                            250
                        );
                    }
                }, 4000);
            }

            window.showDashboardToast = showToast;

            /* =====================================================
               MODULE WAITING

            async function waitForModules(
                timeout = 12000
            ) {
                const started =
                    Date.now();

                while (
                    Date.now() - started <
                    timeout
                ) {
                    const coreReady =
                        window.FirebaseCoreReady;

                    const authReady =
                        window.FirebaseAuthReady;

                    const dbReady =
                        window.FirebaseDatabaseReady;

                    const contentReady =
                        window.FirebaseContentReady;

                    const featuresReady =
                        window.FirebaseFeaturesReady;

                    if (
                        coreReady &&
                        authReady &&
                        dbReady &&
                        contentReady &&
                        featuresReady
                    ) {
                        return true;
                    }

                    await new Promise(
                        resolve =>
                            setTimeout(resolve, 80)
                    );
                }

                return false;
            }

            /* =====================================================
               AUTH / IDENTITY
               ===================================================== */

            function getIdentity() {
                try {
                    if (
                        window.FirebaseAuth &&
                        typeof
                            window.FirebaseAuth.getCurrentUser ===
                            "function"
                    ) {
                        const user =
                            window.FirebaseAuth
                                .getCurrentUser();

                        if (user) {
                            return user;
                        }
                    }
                } catch (error) {
                    console.warn(
                        "FirebaseAuth identity error:",
                        error
                    );
                }

                try {
                    const keys = [
                        "currentUser",
                        "student",
                        "currentStudent"
                    ];

                    for (const key of keys) {
                        const raw =
                            localStorage.getItem(key);

                        if (!raw) {
                            continue;
                        }

                        try {
                            const parsed =
                                JSON.parse(raw);

                            if (parsed) {
                                return parsed;
                            }
                        } catch (_) {
                            return {
                                phone: raw
                            };
                        }
                    }
                } catch (error) {
                    console.warn(
                        "Local identity error:",
                        error
                    );
                }

                return null;
            }

            function hasLoggedInIdentity() {
                const user =
                    getIdentity();

                return !!user;
            }

            async function logout() {
                try {
                    if (
                        window.FirebaseAuth &&
                        typeof
                            window.FirebaseAuth.signOut ===
                            "function"

                        await window.FirebaseAuth
                            .signOut();
                    } else {
                        const keys = [
                            "currentUser",
                            "student",
                            "currentStudent",
                            "currentStudentKey",
                            "firebaseUser"
                        ];

                        keys.forEach(key => {
                            localStorage.removeItem(
                                key
                            );
                        });
                    }
                } catch (error) {
                    console.error(
                        "Logout error:",
                        error
                    );

                    try {
                        const keys = [
                            "currentUser",
                            "student",
                            "currentStudent",
                            "currentStudentKey",
                            "firebaseUser"
                        ];

                        keys.forEach(key => {
                            localStorage.removeItem(
                                key
                            );
                        });
                    } catch (_) {}

                } finally {
                    window.location.replace(
                        "login.html"
                    );
                }
            }

            async function loadSubjects() {
                try {
                    if (
                        window.FirebaseContent &&
                        typeof
                            window.FirebaseContent
                                .getSubjects ===
                            "function"
                    ) {
                        const subjects =
                            await window.FirebaseContent
                                .getSubjects();

                        return Array.isArray(subjects)
                            ? subjects
                            : [];
                    }
                } catch (error) {
                    console.warn(
                        "Subjects load failed:",
                        error
                    );
                }

                try {
                    if (
                        window.FirebaseDatabase &&
                        window.FirebaseDatabase
                            .section
                    ) {
                        const result =
                            await window.FirebaseDatabase
                                .get("subjects");

                        if (
                            result &&
                            typeof result ===
                                "object"
                        ) {
                            return Object.entries(
                                result
                            ).map(
                                ([id, value]) => ({
                                    id,
                                    ...(value || {})
                                })
                            );
                        }
                    }
                } catch (error) {
                    console.warn(
                        "Fallback subjects load failed:",
                        error
                    );
                }

                return [];
            }

                const text =
                    (
                        safeString(
                            subject.id ||
                            subject.name ||
                            subject.title
                        )
                    ).toLowerCase();

                if (
                    text.includes("anatom")
                ) {
                    return "🦴";
                }

                if (
                    text.includes("phys")
                ) {
                    return "🫀";
                }

                if (
                    text.includes("english")
                ) {
                    return "A";
                }

                if (
                    text.includes("fenc") ||
                    text.includes("مبارز")
                ) {
                    return "⚔";
                }

                if (
                    text.includes("psych") ||
                    text.includes("نفس")
                ) {
                    return "🧠";
                }

                if (
                    text.includes("train") ||
                    text.includes("تدريب")
                ) {
                    return "🏃";
                }

                if (
                    text.includes("move") ||
                    text.includes("حرك")
                ) {
                    return "🏋";
                }

                return "▦";
            }

                    let notifications = [];

                    if (
                        window.FirebaseFeatures &&
                        typeof
                            window.FirebaseFeatures
                                .getNotifications ===
                            "function"
                    ) {
                        notifications =
                            await window.FirebaseFeatures
                                .getNotifications();

                        window.FirebaseDatabase

                        notifications =

                                .notifications
                                .getAll();

                        notifications.slice(0, 5)

                        unread.length

                    console.warn(
                        "Notifications load failed:",
                        error
                    );

                    renderNotifications([]);
                    updateNotificationBadge(0);
                }

                count

                    $("#topNotificationDot");

                if (count > 0) {

                    if (badge) {
                        badge.hidden = false;
                        badge.textContent =
                            count > 99
                                ? "99+"
                                : count;
                    }

                    if (dot) {
                        dot.hidden = false;
                    }

                    if (badge) {
                        badge.hidden = true;
                    }

                    if (dot) {

            /* =====================================================
               DATE
               ===================================================== */

            function formatDate(value) {
                if (
                    value === null ||
                    value === undefined ||
                    value === ""
                ) {
                    return "منذ قليل";
                }

                let date;

                if (
                    typeof value === "number"
                ) {
                    date = new Date(value);
                } else if (
                    typeof value === "object" &&
                    value.seconds
                ) {
                    date = new Date(
                        value.seconds * 1000
                    );
                } else {
                    const numeric =
                        Number(value);

                    if (
                        Number.isFinite(
                            numeric
                        ) &&
                        numeric > 100000000000
                    ) {
                        date = new Date(numeric);
                    } else {
                        date = new Date(value);
                    }
                }

                if (
                    !date ||
                    Number.isNaN(
                        date.getTime()
                    )
                ) {
                    return "منذ قليل";
                }

                try {
                    return new Intl.DateTimeFormat(
                        "ar-EG",
                        {
                            year: "numeric",
                            month: "short",
                            day: "numeric"
                        }
                    ).format(date);
                } catch (_) {
                    return date.toLocaleDateString();
                }
            }

            function setupMobileMenu() {
                const sidebar =
                    $("#sidebar");

                const overlay =
                    $("#sidebarOverlay");

                const button =
                    $("#mobileMenuButton");

                if (
                    !sidebar ||
                    !overlay ||
                    !button
                ) {
                    return;
                }

                const openMenu = () => {
                    sidebar.classList.add(
                        "open"
                    );

                    overlay.classList.add(
                        "active"
                    );

                    document.body.classList.add(
                        "menu-open"
                    );
                };

                const closeMenu = () => {
                    sidebar.classList.remove(
                        "open"
                    );

                    overlay.classList.remove(
                        "active"
                    );

                    document.body.classList.remove(
                        "menu-open"
                    );
                };

                button.addEventListener(
                    "click",
                    openMenu
                );

                overlay.addEventListener(
                    "click",
                    closeMenu
                );

                document
                    .querySelectorAll(
                        ".sidebar a"
                    )
                    .forEach(link => {
                        link.addEventListener(
                            "click",
                            closeMenu
                        );
                    });
            }

            /* =====================================================
               REVEAL ANIMATION
               ===================================================== */

            function setupReveal() {
                const elements =
                    $$(".reveal");

                if (!elements.length) {
                    return;
                }

                if (
                    !("IntersectionObserver" in window)
                ) {
                    elements.forEach(
                        element =>
                            element.classList.add(
                                "visible"
                            )
                    );

                    return;
                }

                const observer =
                    new IntersectionObserver(
                        entries => {
                            entries.forEach(
                                entry => {
                                    if (
                                        entry.isIntersecting
                                    ) {
                                        entry.target.classList.add(
                                            "visible"
                                        );

                                        observer.unobserve(
                                            entry.target
                                        );
                                    }
                                }
                            );
                        },
                        {
                            threshold: 0.08
                        }
                    );

                elements.forEach(
                    element =>
                        observer.observe(
                            element
                        )
                );
            }

                        typeof
                            window.FirebaseAuth
                                .onAuthStateChanged ===
                            "function"
                    ) {
                        window.FirebaseAuth
                            .onAuthStateChanged(

                                    if (
                                        !user &&
                                        !hasLoggedInIdentity()
                                    ) {
                                        window.location.replace(
                                            "login.html"
                                        );
                                    }
                                }
                            );
                    }
                } catch (error) {
                    console.warn(
                        "Auth state listener error:",
                        error
                    );
                }
            }

            /* =====================================================
               MAIN INIT
               ===================================================== */

            async function initDashboard() {

                setYear();

                setupMobileMenu();
                setupReveal();

                const modulesReady =
                    await waitForModules();

                if (!modulesReady) {
                    hideLoader();

                    showToast(
                        "تعذر تحميل مكونات المنصة. حاول تحديث الصفحة.",
                        "error"
                    );

                    return;
                }

                const identity =
                    getIdentity();

                if (!identity) {
                    window.location.replace(
                        "login.html"
                    );

                    return;
                }

                const student =
                    await loadStudent();

                if (student) {
                    renderStudent(
                        student
                    );
                }

                setupAuthState();

                const logoutButton =
                    $("#logoutButton");

                if (logoutButton) {
                    logoutButton.addEventListener(
                        "click",
                        logout
                    );
                }

                await Promise.allSettled([
                    loadStatistics(),
                    loadNotifications()
                ]);

                hideLoader();
            }

            /* =====================================================
               START
               ===================================================== */

            if (
                document.readyState ===
                "loading"
            ) {
                document.addEventListener(
                    "DOMContentLoaded",
                    initDashboard,
                    {
                        once: true
                    }
                );
            } else {
                initDashboard();
            }

            /* =====================================================
               PUBLIC API
               ===================================================== */

            window.CapDashboard = {
                init: initDashboard,
                getIdentity,
                loadStudent,
                loadStatistics,
                loadNotifications,
                renderStudent,
                renderCourses,
                showToast,
                logout
            };

        })();
    </script>

</body>
</html>
